\begin{abstract}
For the Lebesgue Clenshaw--Curtis rule on the $N+1$ Chebyshev extrema we
study the Gram matrix $G_{N,n}$ of the rule on polynomials of degree at
most $n$, beyond the half-exactness range $n\le N/2$ in which it is the
identity. Its extreme eigenvalues are the optimal Marcinkiewicz--Zygmund
constants. We prove that all eigenvalues lie below
$\min\{1+(2n/N-1)^2,5/3\}$ for $n<N$, whereas the bound two fails at
$n=N$. At the endpoint degree $n=N-1$ the smallest eigenvalue satisfies
$(\log N)^2\lambda_{\min}\to\pi^4/8$; equivalently, interpolation from
CC-weighted nodal $\ell^2$ data to $L^2$ has norm
$(2\sqrt2/\pi^2+o(1))\log N$. The largest eigenvalue converges to the norm
$\Lambda_{\mathrm{edge}}$ of an explicit endpoint sampling operator, with
$1.0044369\le\Lambda_{\mathrm{edge}}\le3/2$; the lower bound is
computer-assisted. Consequently the Marcinkiewicz--Zygmund defect
$\eta_N=\norm{I-G_{N,N-1}}_2$ satisfies $(\log N)^2(1-\eta_N)\to\pi^4/8$, and
the condition number grows like $(8\Lambda_{\mathrm{edge}}/\pi^4)(\log N)^2$.
We also give explicit strong constants at every fixed degree ratio below one,
and a uniform error factor for CC hyperinterpolation. The proofs use the sign
structure of the Chebyshev mass and defect matrices, a discrete primitive
estimate with a signed sinc remainder, and a weak-compactness argument for
the endpoint operator. The decimal value of $\Lambda_{\mathrm{edge}}$ is
left open.
\end{abstract}

\section{Introduction}\label{sec:context}

Let $Q(f)=\sum_j w_j f(x_j)$ approximate a finite positive measure $\mu$,
and let $G=(Q(P_iP_j))_{ij}$ in a real $L^2(\mu)$-orthonormal basis
of a polynomial space. The basis-independent quantity
\begin{equation}\label{eq:eta}
 \eta=\norm{I-G}_2
 =\max\{1-\lamin(G),\lamax(G)-1\}
\end{equation}
is the smallest nonnegative constant in the Marcinkiewicz--Zygmund (MZ)
inequality
\[
 |Q(p^2)-\norm p_{L^2(\mu)}^2|\le\eta\norm p_{L^2(\mu)}^2 .
\]
In particular, $\eta<1$ means that every Gram eigenvalue lies in $(0,2)$.
If $Q$ is exact through degree $d$, exactness alone gives $G=I$ through
degree $\lfloor d/2\rfloor$, the \emph{half-exactness range}. Beyond it the
constants carry information about the rule. An, Sommariva and Vianello (ASV)
study their role in polynomial approximation and examine them numerically
for several rules, including Clenshaw--Curtis (CC) on the interval
\cite{LIT-045}. They distinguish degree-dependent weak constants from strong
constants uniform along a sampling regime, and they report the CC behaviour
beyond half-exactness as not known theoretically. This paper answers that
question, up to the decimal value of one constant.

\paragraph{Setting}
Throughout,
\begin{equation}\label{eq:nodes}
 N\ge2,\qquad \theta_j=\frac{j\pi}{N},\quad x_j=\cos\theta_j,
 \quad 0\le j\le N,
 \qquad Q_N(f)=\sum_{j=0}^N w_j f(x_j),
\end{equation}
where $w_j=\int_{-1}^1\ell_j(x)\dd x$ and $\ell_j\in\Pn_N$ is
the cardinal polynomial at $x_j$. The measure is unnormalized Lebesgue
measure $\dd x$. Write $G_{N,n}$ for the Gram matrix on $\Pn_n$ and
$\lambda_\pm(N,n)$ for its extreme eigenvalues, so that
\begin{equation}\label{eq:variational}
 \lambda_-(N,n)=\inf_{0\ne p\in\Pn_n}
 \frac{Q_N(p^2)}{\int_{-1}^1p^2},\qquad
 \lambda_+(N,n)=\sup_{0\ne p\in\Pn_n}
 \frac{Q_N(p^2)}{\int_{-1}^1p^2}.
\end{equation}
The node index $N$, the point count $N+1$ and the maximal algebraic degree
of exactness $d_N$ are distinct:
\begin{equation}\label{eq:ade}
 d_N=\begin{cases}N,&N\text{ odd},\\N+1,&N\text{ even}.
 \end{cases}
\end{equation}
(ASV use an exactness-degree parameter; An and Wu's CC parameter $m$ counts
points \cite{LIT-046}.) Thus $G_{N,n}=I$ for $n\le N/2$, and the interesting
range is $N/2<n\le N$. Natural logarithms are used throughout.

\subsection{Main results}

The first result concerns the whole range below full degree.

\begin{theorem}[Upper spectrum]\label{thm:intro-upper}
For $0\le n<N$, with $\rho_{N,n}=\max\{2n/N-1,0\}^2$,
\begin{equation}\label{eq:main-upper}
 0<\lambda_-(N,n)\le\lambda_+(N,n)\le\min\Bigl\{1+\rho_{N,n},\,\frac53\Bigr\}<2 .
\end{equation}
At full degree, $\lambda_+(N,N)>2$.
\end{theorem}

This is Theorems~\ref{thm:ratio-upper} and~\ref{thm:five-thirds} and
Remark~\ref{rem:sign-boundary}. At the endpoint degree $n=N-1$ the two
extreme eigenvalues are determined asymptotically.

\begin{theorem}[Endpoint minimum]\label{thm:intro-min}
Through all integers $N$,
\begin{equation}\label{eq:main-endpoint}
 \lim_{N\to\infty}(\log N)^2\lambda_-(N,N-1)
 =\lim_{N\to\infty}(\log N)^2\lambda_-(N,N)=\frac{\pi^4}8 .
\end{equation}
Equivalently, interpolation at the nodes, from data normed by
$(\sum_kw_ku_k^2)^{1/2}$ to $L^2(-1,1)$, has norm
$(2\sqrt2/\pi^2+o(1))\log N$. Moreover
$K_N^{-1}\le\lambda_-(N,N-1)\le128000/(\log N)^2$ for every $N\ge2$, with
$K_N=O((\log N)^2)$ explicit in \eqref{eq:KN}.
\end{theorem}

This is Theorems~\ref{thm:sharp-min}, \ref{thm:cardinal}
and~\ref{thm:harmonic} and Corollary~\ref{cor:interp-norm}.

\begin{theorem}[Endpoint maximum]\label{thm:intro-max}
Through all integers $N$, $\lambda_+(N,N-1)\to\Lambda_{\mathrm{edge}}$, where
$\Lambda_{\mathrm{edge}}$ is the squared norm \eqref{eq:Lambda-edge} of an
explicit endpoint sampling operator, and
$1.0044369\le\Lambda_{\mathrm{edge}}\le3/2$. The lower bound $1.0044369$ is
computer-assisted.
\end{theorem}

This is Theorem~\ref{thm:edge-limit} and
Proposition~\ref{prop:edge-certified}. Combining the three theorems gives the
MZ defect and the condition number at the endpoint.

\begin{corollary}\label{cor:intro}
Let $\eta_N=\norm{I-G_{N,N-1}}_2$.
\begin{enumerate}
\item $1-\eta_N=\lambda_-(N,N-1)$ for every $N\ge57$; hence
$(\log N)^2(1-\eta_N)\to\pi^4/8$, and
$\kappa_2(G_{N,N-1})/(\log N)^2\to8\Lambda_{\mathrm{edge}}/\pi^4$
(Corollary~\ref{cor:endpoint}).
\item For every $0\le\theta<1$ and $0\le n\le\theta N$, the spectrum of
$G_{N,n}$ lies in $[(1-\theta)^2/C_*,\,1+\max\{2\theta-1,0\}^2]$ with
$C_*=12(1+\pi)^4(1+2/\pi)$ (Corollary~\ref{cor:strong}).
\item The CC hyperinterpolant of degree $n<N$ satisfies
$\norm{f-\mathcal L_nf}_{L^2}\le(1+\sqrt{5/3})\sqrt2\,E_k(f)$ under the
reproduction condition of Section~\ref{sec:hyper}.
\end{enumerate}
\end{corollary}

The constants $5/3$ and $C_*$ are not sharp. The constants $\pi^4/8$ and
$\Lambda_{\mathrm{edge}}$ are sharp: they are the limits.

\paragraph{Methods}
The upper bounds use the sign structure of the defect between continuous and
discrete Chebyshev mass matrices (Section~\ref{sec:upper}). The endpoint
minimum is a converse MZ problem at a borderline. Near an endpoint, on the
scale $\theta=\pi t/N$, interpolation acts like a discrete Hilbert transform.
Its leading part is a discrete primitive on the logarithmic scale, which is
controlled by the norm $2/\pi$ of the Volterra operator; the remainder is
controlled with its signs kept (Section~\ref{sec:endpoint},
Appendix~\ref{app:sharp}). The endpoint maximum is decided on the microscopic
scale $\theta=\xi/N$: a compact transition part converges, and a positive
reflected part is only lower semicontinuous (Section~\ref{sec:edge-max},
Appendix~\ref{app:edge}). The fixed-ratio lower bound uses a filtered
Fej\'er reconstruction (Section~\ref{sec:ratio}).

\paragraph{Outline}
Section~\ref{sec:prelim} collects exact CC identities and weight bounds.
Sections~\ref{sec:upper}--\ref{sec:consequences} prove the results above;
Section~\ref{sec:ratio} treats fixed degree ratios, and
Section~\ref{sec:hyper} hyperinterpolation and an assumption of An and Wu.
Section~\ref{sec:numerics} lists measurements and open problems. The
appendices contain the longer proofs and the certificate.

\subsection{Relation to the literature}

\paragraph{Attribution}
The folded Chebyshev moments used below are established prior algebra:
Trefethen's Theorem~5.2 and formulas (5.2)--(5.4) give the mode aliasing
\cite{LIT-055}; selected coefficient-decay results of
Xiang--Bornemann provide related quadrature context \cite{LIT-056}.
ASV equation~(7) already uses the upper-MZ factor
$1+\sqrt{1+\eta}$, together with an aliasing term, and their
equation~(8) separates the inverse-Gram least-squares construction
\cite{LIT-045}. The generic upper-Gram hyperinterpolation mechanism
in Section~\ref{sec:hyper} is consequently known context; its proof is
included to identify precisely the CC input and the required reproduction.

\paragraph{Converse MZ inequalities}
Writing $\mathcal I_N$ for interpolation at the $N+1$ nodes, positivity of
the weights and uniqueness of interpolation give
\[
 \sup_{0\ne p\in\Pn_n}\frac{\int_{-1}^1p^2}{Q_N(p^2)}=\frac1{\lambda_-(N,n)},
\]
i.e.\ $\lambda_-(N,n)^{-1/2}$ is the norm of $\mathcal I_N$ from nodal data of
degree $\le n$ with the CC-weighted $\ell^2$ norm to $L^2(\dd x)$.
Moreover, $\tfrac12\Delta_N(x_j)\le w_j\le5\Delta_N(x_j)$ with
$\Delta_N(x)=\sqrt{1-x^2}/N+N^{-2}$ (from Lemma~\ref{lem:weight-comparison} and the endpoint weights $w_0=w_N\in\{N^{-2},(N^2-1)^{-1}\}$
below), so CC weights may be exchanged with Christoffel-type weights at a
cost of absolute constants. In this language our endpoint result concerns
a borderline converse Marcinkiewicz--Zygmund inequality.
We compare with the following sources.
\begin{itemize}
\item Sloan--Smith's 1978 CC product-integration paper \cite{LIT-047}
controls the sum of absolute product weights, i.e.\ a $C[-1,1]$ functional
norm. Its companion \cite{LIT-048} develops heuristic error estimates and
numerical tests and states no theorem. In the portions inspected we found
no quadratic-form or nodal $\ell^2$ estimate in either paper.
\item The weighted mean-convergence results of Erd\H{o}s--Feldheim,
Askey, Nevai, and Smith's Lobatto theorem
\cite{LIT-061,LIT-062,LIT-063,LIT-065} bound interpolation from
\emph{uniform} data into weighted $L^p$. Nevai's quasi-Lagrange theorem
includes the CC nodes. Uniform data form a stronger input norm than
weighted nodal $\ell^2$, so these results are consistent with, and do not
imply, logarithmic growth of the $\ell^2\to L^2$ norm.
\item Lubinsky's converse inequalities on the full interpolation space
\cite{LIT-064} require an exponent in an open range. For CC nodes,
Lebesgue measure and $p=2$, the nodal polynomial
$\sin\theta\sin N\theta$ sits exactly at the excluded endpoint.
\item Mastroianni, Milovanovi\'c and Notarangelo \cite{LIT-299}
characterize, for interpolation at the zeros of $p_m((1-x^2)^\alpha)$
together with $\pm1$, when the weighted $L^p$ norm of $P\in\Pn_{m+1}$ is
equivalent to a nodal sum plus an endpoint term. With $\alpha=1/2$ these
are the CC nodes ($N=m+1$). For Lebesgue measure and $p=2$ their strict
condition fails at its endpoint. Since the nodal side of their equivalence
is comparable to $Q_N(p^2)$ by the weight comparison above, the necessity
part of their criterion yields $\inf_N\lambda_-(N,N)=0$ on the full space
$\Pn_N$. This is classical, but it is only a statement about a
subsequence on $\Pn_N$, with no rate.
\item Mastroianni--Russo \cite{LIT-293} and Xu \cite{LIT-296} treat
converse inequalities at Gauss nodes without endpoint terms, and
K\"onig--Nielsen \cite{LIT-294} treat same-weight Jacobi nodes. At the
zeros of $U_{N-1}$ with Lebesgue measure and $p=2$, the Mastroianni--Russo
condition requires $(1-x^2)^{-1/2}\in L^2$, which fails logarithmically.
These are analogous borderline phenomena; we do not derive the CC
statements from them.
\item Mastroianni and Russo's extended converse theorem \cite[Thm.~11]{LIT-511}
covers the CC nodes for even $N$ on $\Pn_N$. For Lebesgue measure and $p=2$
its $A_2$ hypothesis fails exactly at the power-weight endpoint, and the
paper states no rate for the lower constant; its forward theorem gives
boundedness of the upper constant, as does \cite{LIT-050}.
\item The exact CC evaluation in Lemma~\ref{lem:U-exact},
$Q_N(U_{N-1}^2)=O(1)$ while $\int U_{N-1}^2\asymp\log N$ (an elementary
computation), gives the single-$U$ upper bound
$\lambda_-(N,N-1)=O(1/\log N)$ and hence vanishing for every $N$. The order
$(1+\log N)^{-2}$ requires the separate harmonic witness of
Theorem~\ref{thm:harmonic} for the upper bound and Theorem~\ref{thm:cardinal}
for the lower bound.
\item The algebraic Marcinkiewicz inequality of Mastroianni--Totik
\cite{LIT-051} applies at CC nodes only when $n\le N/(\pi M)$ for their
unspecified constant $M$. For every admissible $M\ge2/\pi$ this lies within
the half-exactness range $n\le N/2$, where exactness already gives
$\lambda_\pm(N,n)=1$. In its documented form it therefore adds nothing
beyond exactness for the questions of Section~\ref{sec:ratio}.
\item The forward quadrature-sum theorem of Lubinsky--M\'at\'e--Nevai
\cite{LIT-050} applies at CC nodes, for each fixed $l$, to every degree
$\le lN$. It gives $\lambda_+(N,n)\le C_l$ with an unspecified constant
depending on $l$. The threshold
$2$ and the explicit constants of Theorem~\ref{thm:ratio-upper} and its
endpoint corollary are not implied.
\item The later Sloan--Smith papers \cite{LIT-297,LIT-298} relate
product-integration convergence to mean convergence with uniform data.
\item The interpolation book of Mastroianni--Milovanovi\'c
\cite{LIT-049} (chapters 2, 4 and 5),
Widom's and Widom--Wilf's Hankel asymptotics \cite{LIT-053,LIT-054},
the Chebyshev--Legendre method \cite{LIT-058} and recent
Chebyshev--Lobatto interpolation estimates in the Legendre norm
\cite{LIT-295} concern uniform norms, Sobolev inputs or different matrix
problems. Lubinsky's survey \cite{LIT-052} records genuine converse
quadrature-sum (nodal-norm) theorems, but at Gauss nodes and weights.
\end{itemize}
We found no statement of the logarithmic order of $\lambda_-(N,N-1)$, of
its limit constant, of the limit of $\lambda_+(N,N-1)$, of an all-$N$ upper
constant below two, or of explicit fixed-ratio constants; our search was
documented but not exhaustive. The published version of \cite{LIT-045}
states that the CC behaviour beyond half-exactness is, to its authors'
knowledge, not known theoretically. The arguments below do not depend on
any of these sources.

\section{Preliminaries}\label{sec:prelim}

Put $T_k(\cos\theta)=\cos(k\theta)$ and
\[
 \mu_k=\int_{-1}^1T_k(x)\dd x
 =\begin{cases}0,&k\text{ odd},\\2/(1-k^2),&k\text{ even}.
 \end{cases}
\]
The moment formula follows directly by integrating
$\sin\theta\cos(k\theta)$; $k=0,1$ can be checked separately.

\begin{lemma}[Exact identities]\label{lem:identities}% IDs: PD-T013
Set $\gamma_0=\gamma_N=1/2$ and $\gamma_j=1$ otherwise. Then
\begin{equation}\label{eq:weights}
 w_j=\frac{2\gamma_j}{N}\left[
 1-\sum_{r=1}^{\lfloor(N-1)/2\rfloor}
      \frac{2\cos(2r\theta_j)}{4r^2-1}
 -\boldsymbol1_{2\mid N}\frac{\cos(N\theta_j)}{N^2-1}
 \right]>0.
\end{equation}
The endpoint weights equal $N^{-2}$ for odd $N$ and $(N^2-1)^{-1}$
for even $N$. For $0\le k\le2N$,
\begin{equation}\label{eq:alias}
 Q_N(T_k)=\begin{cases}\mu_k,&k\le N,\\\mu_{2N-k},&N<k\le2N.
 \end{cases}
\end{equation}
For $0\le i,j\le n\le N$, the continuous and discrete Chebyshev
mass matrices satisfy
\begin{equation}\label{eq:mass}
\begin{gathered}
 M_{ij}=\frac{\mu_{|i-j|}+\mu_{i+j}}2,\quad
 R_{ij}=\frac{\mu_{|i-j|}+Q_N(T_{i+j})}2,\\
 D:=2(M-R),\quad D_{ij}=\mu_{i+j}-Q_N(T_{i+j}).
\end{gathered}
\end{equation}
For $n<N$, $D$ is entrywise nonnegative, with positive entries precisely
when $i+j>N$ is even.
\end{lemma}
\begin{proof}
Finite geometric sums give
$\sum_{j=0}^N\gamma_j\cos(k\theta_j)\cos(l\theta_j)
=N\delta_{kl}/(2\gamma_k)$ for $0\le k,l\le N$.
Indeed the corresponding weighted cosine sum vanishes at indices
$1,\ldots,2N-1$ and is $N$ at $0,2N$; the product formula gives the claim.
Thus
\[
 \ell_j(\cos\theta)=\frac{2\gamma_j}{N}
       \sum_{k=0}^N\gamma_k\cos(k\theta_j)\cos(k\theta).
\]
Integration proves \eqref{eq:weights}. The telescoping identity
\begin{equation}\label{eq:telescope}
 \sum_{r=1}^q\frac2{4r^2-1}=1-\frac1{2q+1}
\end{equation}
shows that its bracket is at least $1/N$ for odd $N$ and
$N/(N^2-1)$ for even $N$. It also proves the endpoint formulas.
Cardinal interpolation gives exactness on $\Pn_N$, and
$T_k(x_j)=T_{2N-k}(x_j)$ gives \eqref{eq:alias}.
The identity $T_iT_j=(T_{i+j}+T_{|i-j|})/2$ proves \eqref{eq:mass}.
If $s=i+j>N$ is even and $n<N$, then $r=2N-s$ satisfies $2\le r<s$,
so $D_{ij}=2/(r^2-1)-2/(s^2-1)>0$. Odd moments vanish.
Finally symmetry gives exactness on degree $N+1$ when $N$ is even.
For odd $N$ the moment $N+1$ aliases to $N-1$ and differs from it;
for even $N$ the moment $N+2$ aliases to $N-2$ and differs from it.
This also proves the maximality assertion \eqref{eq:ade}.
\end{proof}

\begin{lemma}[Coefficient mass floor]\label{lem:floor}% IDs: PD-T015
For $n=N-1$ and $E=\diag(2,1,\ldots,1)$,
\begin{equation}\label{eq:floor}
 \frac1N E\preceq M\preceq\frac\pi2 E.
\end{equation}
\end{lemma}
\begin{proof}
The off-diagonal entries of $M$ are nonpositive. Let $b_0=1/2$,
$b_i=1$ for $i\ge1$, and let $m_i$ be the least integer at least $N$
having the parity of $i$. Put $k_i=m_i-1\in\{N-1,N\}$. Telescoping yields
\begin{equation}\label{eq:rowfloor}
 (Mb)_0=\frac1{k_0},\qquad
 (Mb)_i=\frac12\left(\frac1{k_i-i}+\frac1{k_i+i}\right)
       =\frac{k_i}{k_i^2-i^2}\quad(i\ge1).
\end{equation}
To verify these finite row sums, note that the absolutely convergent
even moment series obeys
$\sum_{h\in2\mathbb Z}\mu_{|h|}=2-4\sum_{r\ge1}(4r^2-1)^{-1}=0$.
The infinite row weighted by $b$ is half that sum and is zero.
Its omitted tail, beginning at $j=m_i$ in steps of two, is
\[
 -\sum_{j=m_i,m_i+2,\ldots}
 \left[\frac1{(j-i)^2-1}+\frac1{(j+i)^2-1}\right].
\]
Using $1/(u^2-1)=[1/(u-1)-1/(u+1)]/2$ gives
\eqref{eq:rowfloor}, including $i=0$. Hence
$(Mb)_0/b_0\ge2/N$ and $(Mb)_i/b_i\ge1/N$ for $i\ge1$.
For a real vector $a$, direct expansion gives
\[
 a^TMa=\sum_i\frac{(Mb)_i}{b_i}a_i^2
 +\sum_{i<j}(-M_{ij})b_i b_j
             \left(\frac{a_i}{b_i}-\frac{a_j}{b_j}\right)^2.
\]
This proves the lower bound. If $q(\theta)=\sum_i a_i\cos(i\theta)$,
then $a^TMa=\int_0^\pi q^2\sin\theta\dd\theta
\le\int_0^\pi q^2\dd\theta=(\pi/2)a^TEa$, proving the upper bound.
\end{proof}

\begin{lemma}[Weight comparison]\label{lem:weight-comparison}
For $1\le j\le N-1$,
\begin{equation}\label{eq:weightcomp}
 \frac{\sin\theta_j}{N}\le w_j\le\frac{5\sin\theta_j}{N}.
\end{equation}
If $m_j=1+\min(j,N-j)$, then $w_j\ge m_j/N^2$ for all $0\le j\le N$.
\end{lemma}
\begin{proof}
The absolutely convergent cosine series
\[
 \frac\pi2\sin\theta=1-2\sum_{r\ge1}
                  \frac{\cos(2r\theta)}{4r^2-1}\quad(0\le\theta\le\pi)
\]
follows by integrating against cosine modes; absolute convergence and
uniqueness of the cosine expansion identify the continuous function.
For example, uniqueness follows from density of polynomials in
$\cos\theta$ among continuous functions on $[0,\pi]$.
Let $S_N(\theta)$ denote the bracket in \eqref{eq:weights}.
Telescoping its tail gives
\[
 |S_N(\theta)-(\pi/2)\sin\theta|
 \le\begin{cases}1/N,&N\text{ odd},\\N/(N^2-1),&N\text{ even}.
 \end{cases}
\]
At interior nodes $\sin\theta_j\ge2/N$. The error is therefore at most
$(2/3)\sin\theta_j$ for all $N\ge2$, including the even worst case $N=2$.
Since $w_j=2S_N(\theta_j)/N$ and
$\pi/2-2/3\ge1/2$, $\pi+4/3<5$, this proves
\eqref{eq:weightcomp}. Moreover
$\sin\theta_j\ge2\min(j,N-j)/N$, so
$w_j\ge2\min(j,N-j)/N^2\ge m_j/N^2$ in the interior.
The endpoint formulas prove the remaining cases.
\end{proof}

\section{The upper spectrum}\label{sec:upper}

The upper eigenvalues are controlled by the sign structure of the defect
matrix $D$ of Lemma~\ref{lem:identities}. Theorem~\ref{thm:upper} gives a
threshold below two for every $N$; Theorem~\ref{thm:ratio-upper} refines it
at each degree ratio, and Theorem~\ref{thm:five-thirds} gives a bound
uniform in $N$.

\begin{theorem}[Strict threshold and endpoint refinement]\label{thm:upper}% IDs: PD-T014, PD-T022
For $0\ne p\in\Pn_{N-1}$,
\[
 0<Q_N(p^2)\le\left(2-\frac4N+\frac4{N^2}\right)\int_{-1}^1p^2
 <2\int_{-1}^1p^2.
\]
The same bound holds on every $\Pn_n$ with $n<N$.
For $N=2$ every quotient equals one; for $N\ge3$ the upper inequality is strict.
\end{theorem}
\begin{proof}
Set $A=2M-R=M+D/2$. We show that its off-diagonal entries are
nonpositive. Opposite parities give zero. For distinct equal-parity indices,
write $d=|i-j|\ge2$ and $s=i+j$. If $s\le N$, $A_{ij}=M_{ij}<0$.
Otherwise let $r=2N-s$. Since $i,j\le N-1$,
$r-d=2(N-\max(i,j))\ge2$, and
\begin{equation}\label{eq:zentry}
 A_{ij}=-\frac1{d^2-1}-\frac2{s^2-1}+\frac1{r^2-1}<0.
\end{equation}
Write $p=\sum_i a_iT_i$ and $v_i=|a_i|$. The off-diagonal signs imply
$a^TAa\ge v^TAv$: diagonal terms agree and $a_i a_j\le v_i v_j$.
Since $D$ is entrywise nonnegative,
\[
 2\int p^2-Q_N(p^2)\ge v^TMv+\tfrac12v^TDv\ge v^TMv.
\]
This is positive for $v\ne0$, proving the qualitative threshold without the coefficient floor. Positivity of $Q_N(p^2)$ follows from
positive weights and the impossibility that a nonzero polynomial of
degree at most $N$ vanish at all $N+1$ nodes. Restriction proves the rest.

The explicit endpoint refinement is the $n=N-1$ corollary of the independent
degree-sensitive sign comparison in Theorem~\ref{thm:ratio-upper}:
$\rho=(1-2/N)^2$ and
$1+\rho=2-4/N+4/N^2$. At $N=2$, $\rho=0$ and exactness applies;
at $N\ge3$ the strict case of that theorem applies.
\end{proof}

\begin{theorem}[Degree-sensitive upper estimate]\label{thm:ratio-upper}% IDs: PD-T022
For $0\le n<N$, put $\rho=\max\{2n/N-1,0\}^2$. Then
\[
 Q_N(p^2)\le(1+\rho)\int_{-1}^1p^2\qquad(p\in\Pn_n).
\]
If $2n\le N$, every quotient equals one. If $2n>N$, the displayed
inequality is strict for every nonzero $p$.
\end{theorem}
\begin{proof}
Exactness settles $2n\le N$. Otherwise set $t=2n/N-1>0$,
$\rho=t^2$, and $A_\rho=(1+\rho)M-R=\rho M+D/2$ on indices $0,\ldots,n$.
Opposite-parity off-diagonal entries vanish; equal-parity unaliased entries
are negative. For an aliased equal-parity pair let
$s=i+j>N$, $d=|i-j|\ge2$, $r=2N-s$, $\delta=N-n>0$.
Then $r<N$, $r-d\ge2\delta$, and
\[
 \frac dr\le1-\frac{2\delta}{N}=t,\qquad
 \frac{d^2-1}{r^2-1}\le\frac{d^2}{r^2}\le\rho.
\]
Consequently
\[
 (A_\rho)_{ij}=-\frac\rho{d^2-1}
                -\frac{1+\rho}{s^2-1}+\frac1{r^2-1}<0.
\]
For this matrix with nonpositive off-diagonal entries, replacement of
coefficients $a$ by $v=|a|$ gives
$a^TA_\rho a\ge\rho v^TMv+v^TDv/2>0$ for $a\ne0$.
This proves the result.
\end{proof}

\begin{corollary}[Endpoint upper constant]\label{cor:ratio-endpoint}% IDs: PD-T022
For every $N\ge2$,
\[
 \lambda_+(N,N-1)\le1+(1-2/N)^2=2-\frac4N+\frac4{N^2}.
\]
Equality holds at $N=2$; the displayed upper inequality is strict for $N\ge3$.
\end{corollary}
\begin{proof}
Set $n=N-1$ in Theorem~\ref{thm:ratio-upper}. Then
$\max\{2n/N-1,0\}=1-2/N$. The case $N=2$ has $2n=N$ and $\rho=0$;
every $N\ge3$ has $2n>N$ and $\rho>0$. This proves the endpoint
refinement used in Theorem~\ref{thm:upper}.
\end{proof}

% Source: PD-T526; topics/padua-mz-sharp-constants-20261106/theory/S2-uniform-upper.md
\begin{theorem}[A uniform upper bound]\label{thm:five-thirds}
For every $N\ge2$ and every nonzero $p\in\Pn_{N-1}$,
\[
 0<Q_N(p^2)\le\frac53\int_{-1}^1p^2 .
\]
Consequently, for all $0\le n<N$,
\[
 \lambda_+(N,n)\le\min\Bigl\{1+\rho_{N,n},\,\frac53\Bigr\},
 \qquad \rho_{N,n}=\max\{2n/N-1,0\}^2 .
\]
\end{theorem}

The bound $5/3$ is not sharp; its point is that it is uniform in $N$,
whereas the degree-sensitive bound of Corollary~\ref{cor:ratio-endpoint}
tends to two. The proof keeps the signs of the defect, but splits it
differently from Theorem~\ref{thm:upper}.

\begin{proof}
Write $p=\sum_{i=0}^{N-1}a_iT_i$, so that $a^TMa=\int p^2$ and
$a^TRa=Q_N(p^2)$ with the matrices of Lemma~\ref{lem:identities} on
the indices $0,\dots,N-1$. For a pair $(i,j)$ put $s=i+j\le2N-2$ and
$r=2N-s\ge2$, and define
\[
 Y_{ij}=\frac{\boldsymbol1_{s\ \mathrm{even}}}{r^2-1},\qquad
 Z_{ij}=\begin{cases}
   Y_{ij},&s\le N,\\[2pt]
   \dfrac{\boldsymbol1_{s\ \mathrm{even}}}{s^2-1},&s>N.
 \end{cases}
\]
We claim the exact identity
\begin{equation}\label{eq:RMHK}
 R=M-Y+Z .
\end{equation}
If $s\le N$, then $Q_N(T_s)=\mu_s$ by \eqref{eq:alias}, so $R_{ij}=M_{ij}$,
and $Y_{ij}=Z_{ij}$. If $s>N$ is even, then $Q_N(T_s)=\mu_r$ and
\eqref{eq:mass} gives
$(M-R)_{ij}=(\mu_s-\mu_r)/2=1/(r^2-1)-1/(s^2-1)=Y_{ij}-Z_{ij}$.
If $s$ is odd, every entry of $M-R$, $H$ and $K$ vanishes.

The matrix $Y$ is positive semidefinite. Indeed, for $i,j\le N-1$,
\[
 \frac12\int_0^1 x^{2N-2-i-j}(1-x^2)\dd x
 =\frac12\Bigl(\frac1{r-1}-\frac1{r+1}\Bigr)=\frac1{r^2-1},
\]
and $Y$ couples only indices of equal parity, so for real $a$
\[
 a^TYa=\frac12\int_0^1(1-x^2)\Bigl[
 \Bigl(\sum_{i\ \mathrm{even}}a_ix^{N-1-i}\Bigr)^2
 +\Bigl(\sum_{i\ \mathrm{odd}}a_ix^{N-1-i}\Bigr)^2\Bigr]\dd x\ge0 .
\]
No sign property of $Z$ or of $M-R$ as a matrix is used.

Let $u_N=N$ for even $N$ and $u_N=N+1$ for odd $N$. Every nonzero
entry of $Z$ equals $1/(u^2-1)$ for an even integer $u\ge u_N$:
$u=r=2N-s\ge N$ if $s\le N$, and $u=s>N$ otherwise. Hence
$0\le Z_{ij}\le1/(u_N^2-1)$. Each row of $Z$ has at most
$\lceil N/2\rceil$ nonzero entries, since $Z$ couples only indices of
equal parity. Using $2|a_ia_j|\le a_i^2+a_j^2$ and symmetry,
\[
 a^TZa\le\sum_{i,j}Z_{ij}|a_ia_j|
 \le\frac{\lceil N/2\rceil}{u_N^2-1}\sum_ia_i^2 .
\]
Lemma~\ref{lem:floor} gives $\sum_ia_i^2\le a^TEa\le N\,a^TMa$.
Together with \eqref{eq:RMHK} and $Y\succeq0$,
\[
 a^TRa\le a^TMa+a^TZa\le(1+\beta_N)\,a^TMa,
\]
where
\[
 \beta_N=\frac{N\lceil N/2\rceil}{u_N^2-1}
 =\begin{cases}
   \dfrac{N^2}{2(N^2-1)},&N\text{ even},\\[6pt]
   \dfrac{N+1}{2(N+2)},&N\text{ odd}.
 \end{cases}
\]
For even $N\ge2$ the first expression is at most its value $2/3$ at $N=2$;
for odd $N\ge3$ the second is below $1/2$. This proves the upper
bound, and positivity follows as in Theorem~\ref{thm:upper}.
Restriction to $\Pn_n$ and Theorem~\ref{thm:ratio-upper} give the
second display.
\end{proof}

Since $\beta_N\to1/2$, the proof also gives
$\limsup_{N\to\infty}\lambda_+(N,N-1)\le3/2$; Section~\ref{sec:edge-max}
identifies the limit.

\begin{remark}[An alternative quantitative proof]% IDs: PD-T015
The mass inequality of Lemma~\ref{lem:floor}
also gives
$v^TMv\ge N^{-1}v^TEv=N^{-1}a^TEa\ge2(\pi N)^{-1}a^TMa$
in the preceding sign argument, proving the weaker bound
\[
 Q_N(p^2)\le\left(2-\frac2{\pi N}\right)\int_{-1}^1p^2.
\]
Its constant is superseded by Theorem~\ref{thm:upper}, since
$4-4/N\ge2>2/\pi$ for every $N\ge2$, but the mass inequality is used again
in Theorem~\ref{thm:five-thirds}.
\end{remark}

\begin{remark}[The sign argument and its boundary]\label{rem:sign-boundary}% IDs: PD-N002, PD-T016
Entrywise nonnegativity of $D$ does not mean $D\succeq0$.
For $N\ge5$, take $i=\lfloor N/2\rfloor$ and $j=i+2\le N-1$.
Then $D_{ii}=0$, $D_{ij}>0$, and the corresponding principal determinant
is $-D_{ij}^2<0$.% IDs: PD-N002
 The proof uses the signs of $A$, not
positive semidefiniteness of the defect. Nor can one extend the theorem
to $\Pn_N$: since $T_N(x_j)^2=1$,
\begin{equation}\label{eq:full}
 \frac{Q_N(T_N^2)}{\int_{-1}^1T_N^2}
 =\frac{2(4N^2-1)}{4N^2-2}>2
 .% IDs: PD-T016
\end{equation}
Here $\int T_N^2=1-1/(4N^2-1)$ by the product identity.
\end{remark}

\section{The endpoint minimum eigenvalue}\label{sec:endpoint}

Write $H_s=\sum_{r=1}^s r^{-1}$ and
\begin{equation}\label{eq:KN}
 h_N=2H_{N+1}-1,\qquad
 K_N=4\pi^2(\pi+1)^2h_N
       \bigl(h_N+2H_{\lfloor N/2\rfloor+1}\bigr).
\end{equation}

\begin{theorem}[Weighted cardinal lower estimate]\label{thm:cardinal}% IDs: PD-T017
For every $p\in\Pn_N$,
\begin{equation}\label{eq:cardinal-result}
 \int_{-1}^1p^2\le K_N Q_N(p^2).
\end{equation}
Consequently $\lambda_-(N,N)\ge K_N^{-1}$, and restriction gives
$\lambda_-(N,N-1)\ge\lambda_-(N,N)\ge K_N^{-1}
\ge c(1+\log N)^{-2}$ for an absolute $c>0$.
\end{theorem}
\begin{proof}
The nodal polynomial $(1-x^2)T_N'(x)$ yields
\[
 \ell_k(\cos\theta)=
 \frac{(-1)^{k+1}\sin\theta\sin(N\theta)}
      {c_kN(\cos\theta-\cos\theta_k)},\qquad
 c_0=c_N=2,\quad c_k=1\ (0<k<N),
\]
with continuous nodal extension. Factor the cosine difference and use
$\sin\theta\le2\sin((\theta+\theta_k)/2)$ and
$|\sin((\theta-\theta_k)/2)|\ge|\theta-\theta_k|/\pi$.
Together with $|\sin(N\theta)|\le\min\{1,N|\theta-\theta_k|\}$,
this gives, for $t=N\theta/\pi$,
\begin{equation}\label{eq:cardinal-decay}
 |\ell_k(\cos\theta)|\le\min\{\pi,|t-k|^{-1}\}
       \le\frac{\pi+1}{1+|t-k|}.
\end{equation}
The first bound includes the continuous value at $t=k$.
Partition $[0,\pi]$ into cells $E_j$ defined by $|t-j|\le1/2$,
truncated at the endpoints. On $E_j$,
\begin{equation}\label{eq:cell}
 |\ell_k(\cos\theta)|\le\frac{D_0}{1+|j-k|},\quad D_0=2(\pi+1),
 \qquad \int_{E_j}\sin\theta\dd\theta\le\frac{\pi^2m_j}{N^2}.
\end{equation}
For the integral, use cell length at most $\pi/N$ and
$\sin\theta\le\pi m_j/N$.

Put $A_{jk}=(1+|j-k|)^{-1}$, $B_{jk}=\sqrt{m_j/m_k}\,A_{jk}$,
and $u_j=m_j^{-1/2}$. Because $m_j\le m_k+|j-k|$,
\begin{align*}
 \sum_k B_{jk}\frac{u_k}{u_j}
 &=m_j\sum_k\frac{A_{jk}}{m_k}
 \le\sum_k A_{jk}+\sum_k\frac1{m_k}
 \le h_N+2H_{\lfloor N/2\rfloor+1},\\
 \sum_j B_{jk}\frac{u_j}{u_k}&=\sum_j A_{jk}\le h_N.
\end{align*}
The harmonic bounds deliberately overcount distances near the endpoints.
The weighted Schur inequality therefore gives
\begin{equation}\label{eq:schur-discrete}
 \norm B_2^2\le h_N\bigl(h_N+2H_{\lfloor N/2\rfloor+1}\bigr).
\end{equation}
For completeness, if the weighted row and column bounds are $R,S$,
then Cauchy--Schwarz gives
$|\sum_kB_{jk}z_k|^2\le(\sum_kB_{jk}u_k)
\sum_kB_{jk}|z_k|^2/u_k$; summing in $j$ gives $RS\sum_k|z_k|^2$.
This proves the stated version without an interpolation theorem.

Set $v_k=p(x_k)$. Cardinal interpolation, \eqref{eq:cell}, and
\eqref{eq:schur-discrete} give
\begin{align*}
 \int p^2
 &\le\frac{\pi^2D_0^2}{N^2}
       \sum_jm_j\left(\sum_kA_{jk}|v_k|\right)^2\\
 &\le K_N\sum_k\frac{m_k}{N^2}|v_k|^2
 \le K_NQ_N(p^2).
\end{align*}
The last inequality is Lemma~\ref{lem:weight-comparison}.
Finally $H_s\le1+\log s$ gives the asserted order.
\end{proof}

The matching upper estimate needs a test polynomial of degree $N-1$.
The simplest is the second-kind Chebyshev polynomial
$U_{N-1}(\cos\theta)=\sin(N\theta)/\sin\theta$, which vanishes at every
interior node.

% Source: PD-T019 (formulas); tracks/mz-constants/theory/m2-u-witness.md
\begin{lemma}[An exact evaluation]\label{lem:U-exact}% IDs: PD-T019
For $S_N=\sum_{j=1}^N(2j-1)^{-1}$,
\begin{equation}\label{eq:Uexact}
 \int_{-1}^1U_{N-1}^2=2S_N,\qquad
 Q_N(U_{N-1}^2)=
 \begin{cases}2,&N\text{ odd},\\2N^2/(N^2-1),&N\text{ even}.
 \end{cases}
\end{equation}
Consequently $\lambda_-(N,N-1)\le q_N$, where $q_N=1/S_N$ for odd $N$ and
$q_N=N^2/[(N^2-1)S_N]$ for even $N$; and $q_N<1/3$ for every $N\ge57$.
\end{lemma}

\begin{proof}
The finite trigonometric product identity
$U_r^2=\sum_{s=0}^rU_{2s}$ follows by summing the geometric sine series.
Also $\int_{-1}^1U_{2s}=\int_0^\pi\sin((2s+1)\theta)\dd\theta
=2/(2s+1)$, giving the continuous formula.
All interior nodal values of $U_{N-1}$ vanish; its endpoint values have
absolute value $N$. The endpoint weights of Lemma~\ref{lem:identities}
prove the discrete formula. Exact rational addition gives
$S_{57}>3$ and $S_{58}>3\cdot58^2/(58^2-1)$. Since $S_N$ increases and
$N^2/(N^2-1)$ decreases, $q_N<1/3$ for all odd $N\ge57$ and all even
$N\ge58$. (For $N=56$ one has $q_N>1/3$.)
\end{proof}

Since $S_N=\frac12\log N+O(1)$, the lemma gives only
$\lambda_-(N,N-1)=O(1/\log N)$. The squared logarithm needs a
combination of many $U_r$. Strictly alternating nodal values with positive
amplitudes would represent a full degree-$N$ interpolant, so we enforce
both the degree and the endpoint cancellation explicitly.

\begin{theorem}[Parity-compatible harmonic witness]\label{thm:harmonic}% IDs: PD-T018
For every $N\ge2$,
\begin{equation}\label{eq:harmonic-result}
 \lambda_-(N,N-1)\le\frac{128000}{(\log N)^2}.
\end{equation}
Together with Theorem~\ref{thm:cardinal}, this gives
$\lambda_-(N,N-1)\asymp(1+\log N)^{-2}$.
\end{theorem}
\begin{proof}
Let $U_r(\cos\theta)=\sin((r+1)\theta)/\sin\theta$.
For $N\ge3$ put $M_0=\lfloor(N-1)/2\rfloor$ and
\begin{equation}\label{eq:witness}
 p_N=\sum_{m=1}^{M_0}\frac{
 U_{N-1-2m}-(1-2m/N)U_{N-1}}m.
\end{equation}
Every term has degree at most $N-1$ and parity $N-1$.
The endpoint values $U_r(\pm1)=(\pm1)^r(r+1)$ show that each
bracket vanishes at both endpoints. Define
\[
 B(\theta)=\sum_{m=1}^{M_0}\frac{\sin(2m\theta)}m,\quad
 A(\theta)=\sum_{m=1}^{M_0}\frac{\cos(2m\theta)}m,\quad
 C(\theta)=A(\theta)-H_{M_0}+\frac{2M_0}{N}.
\]
The sine subtraction formula gives the exact identity
\begin{equation}\label{eq:witness-identity}
 p_N(\cos\theta)=
 \frac{\sin(N\theta)C(\theta)-\cos(N\theta)B(\theta)}{\sin\theta}.
\end{equation}

We record elementary bounds on these finite harmonic sums.
For $0<\theta\le\pi/2$ let $K=\lfloor1/\sin\theta\rfloor$.
A geometric sum bounds
$|\sum_{m=a}^b e^{2im\theta}|\le1/\sin\theta$.
Abel summation bounds a harmonic tail from $K+1$ to any $M_0>K$
by $1/((K+1)\sin\theta)\le1$.
The initial segment satisfies
$|\sum_{m\le K}\sin(2m\theta)/m|\le2K\theta\le\pi$
and $|\sum_{m\le K}\cos(2m\theta)/m|\le H_K
\le1+\log(1/\sin\theta)$; truncate it if $M_0\le K$.
Reflection treats the other half of the interval. Thus
\begin{equation}\label{eq:harmonic-bounds}
 |B(\theta)|\le\pi+1<5,
 \quad A(\theta)\le2+\log\frac{\pi}{2\theta}
 \quad(0<\theta\le\pi/2).
\end{equation}
Since $H_{M_0}\ge\log(M_0+1)$, $M_0+1\ge N/2$, and $2M_0/N\le1$,
\begin{equation}\label{eq:Cbound}
 C(\theta)\le3+\log\pi-\log(N\theta)
              \le5-\log(N\theta).
\end{equation}

At interior nodes \eqref{eq:witness-identity} becomes
$p_N(x_k)=-(-1)^kB(\theta_k)/\sin\theta_k$.
The endpoints are zero. Lemma~\ref{lem:weight-comparison} and
$\sin\theta_k\ge2\min(k,N-k)/N$ imply
\begin{equation}\label{eq:witness-discrete}
 Q_N(p_N^2)\le\frac{125}{N}\sum_{k=1}^{N-1}\frac1{\sin\theta_k}
 \le125H_{\lfloor N/2\rfloor}\le125(1+\log N).
\end{equation}

Write $L=\log N$ and first suppose $L\ge128$.
On $E=[N^{-1/2},1/4]$, \eqref{eq:Cbound} gives
$C\le5-L/2\le-L/4$. The elementary inequality
$(a-b)^2\ge a^2/2-b^2$ applied in \eqref{eq:witness-identity} yields
\[
 \int_{-1}^1p_N^2\ge\frac{L^2}{32}I-25J,\quad
 I=\int_E\frac{\sin^2(N\theta)}{\sin\theta}\dd\theta,\quad
 J=\int_E\frac1{\sin\theta}\dd\theta.
\]
Using $\sin\theta\le\theta$,
\[
 I\ge\tfrac12\log(\sqrt N/4)
       -\tfrac12\left|\int_E\frac{\cos(2N\theta)}\theta\dd\theta\right|.
\]
One integration by parts bounds the last absolute integral by $N^{-1/2}$:
the endpoint terms and the integral of $\theta^{-2}$ sum to
$1/(N\cdot N^{-1/2})$. Hence
$I\ge L/4-\log2-1/(2\sqrt N)\ge L/8$.
Since $\sin\theta\ge\theta/2$ on $E$, $J\le2\log(\sqrt N/4)\le L$.
It follows that
\begin{equation}\label{eq:witness-continuous}
 \int p_N^2\ge\frac{L^3}{256}-25L\ge\frac{L^3}{512},
\end{equation}
where $L^2\ge128^2>12800$ suffices. This also proves $p_N\ne0$.
Equations~\eqref{eq:witness-discrete}--\eqref{eq:witness-continuous}
give a quotient at most $250\cdot512/L^2$.
For $0<L<128$, the constant polynomial gives
$\lambda_-(N,N-1)\le1<128000/L^2$.
This covers every remaining $N$, without an unperformed finite check.
\end{proof}

The witness estimate starts at $\log N\ge128$, but
$128000/(\log N)^2<1$ exactly when
$\log N>\sqrt{128000}\approx357.7709$.
Thus this explicit constant is vacuous throughout all computationally feasible
full-spectrum calculations considered here. The finite values in
Table~\ref{tab:cc} do not exhibit or determine an asymptotic leading constant.

\subsection{The sharp constant}\label{sec:sharp-min}

Theorems~\ref{thm:cardinal} and~\ref{thm:harmonic} fix the order
$(\log N)^{-2}$ but leave the constant open by a factor of order $10^6$.
The exact constant comes from a one-dimensional model. Near the endpoint
$x=1$, write $\theta=\pi t/N$, so that the node $x_k$ corresponds to
$t=k$. For nodal data $u_k=(-1)^ka_k$ supported at $1\le k\ll N$, the
interpolant behaves like $(2N/\pi)\,\sin(\pi t)\,t^{-1}$ times the
partial sum $\sum_{k<t}k^{-1}\cdot(k a_k/N)$, up to a signed error
that is bounded in the relevant norms (Lemma~\ref{lem:signed-remainder}).
On the logarithmic scale $x=\log t/\log N$ the partial sum becomes a
primitive, the CC energy becomes $\pi^2\log N$ times an $L^2$ norm
and the continuous mass becomes $2(\log N)^3$ times the $L^2$ norm of
the primitive; the factor two in the latter is the mean value $1/2$ of
$\sin^2(\pi t)$, counted on each of the two endpoints. The constant is
then $(\pi^2/2)$ times the reciprocal squared norm of the primitive
operator.

\begin{lemma}[The primitive operator]\label{lem:volterra}% IDs: PD-T522
On $L^2(0,1)$ let $(Vf)(x)=\int_0^xf(s)\dd s$. Then $\norm V=2/\pi$, and
\[
 \inf_{0\ne f\in L^2(0,1)}\frac{\pi^2}2\,\frac{\norm f^2}{\norm{Vf}^2}
 =\frac{\pi^4}8,
\]
with equality for $f(x)=\cos(\pi x/2)$.
\end{lemma}

\begin{proof}
Put $F=Vf$ and $\alpha=\pi/2$. For $F\in C^1[0,1]$ with $F(0)=0$,
expanding the square and integrating the cross term by parts give
\[
 \int_0^1|F'-\alpha\cot(\alpha x)F|^2\dd x
 =\int_0^1|F'|^2\dd x-\alpha^2\int_0^1|F|^2\dd x ,
\]
because $(\alpha\cot\alpha x)^2+(\alpha\cot\alpha x)'=-\alpha^2$ and the
boundary term $-\alpha[\cot(\alpha x)|F|^2]_0^1$ vanishes: $\cot\alpha=0$
and $F(x)=O(x)$ at zero. Hence $\norm F^2\le(4/\pi^2)\norm{F'}^2$, and the
inequality extends to every absolutely continuous $F$ with $F(0)=0$ and
$F'\in L^2$ by density. Equality holds for $F(x)=\sin(\alpha x)$, whose
derivative is $\alpha\cos(\alpha x)$.
\end{proof}

\begin{proposition}[An admissible logarithmic trial]\label{prop:sharp-trial}% IDs: PD-T523
Through all integers $N$, and hence along each parity,
\[
 \limsup_{N\to\infty}(\log N)^2\,\lambda_-(N,N-1)\le\frac{\pi^4}8 .
\]
More precisely, let $f\in C^1$ be real, nonzero and compactly supported in
$(0,1)$, let $L=\log N$, and define nodal values
\[
 u_k=\frac{(-1)^kN}{k}\,f\Bigl(\frac{\log k}L\Bigr)\quad(1\le k<N/2),\qquad
 u_{N-k}=(-1)^{N-1}u_k,
\]
with $u_0=u_N=0$ and $u_{N/2}=0$ for even $N$. For all sufficiently
large $N$ the interpolant $p_N$ of these values lies in $\Pn_{N-1}$, is
nonzero, and
\[
 Q_N(p_N^2)=2\pi^2L\norm f^2\,(1+o(1)),\qquad
 \liminf_{N\to\infty}\frac{\int_{-1}^1p_N^2}{L^3}\ge4\norm{Vf}^2 .
\]
\end{proposition}

\begin{theorem}[The sharp endpoint constant]\label{thm:sharp-min}% IDs: PD-T546
Through all integers $N$,
\[
 \lim_{N\to\infty}(\log N)^2\,\lambda_-(N,N-1)
 =\lim_{N\to\infty}(\log N)^2\,\lambda_-(N,N)=\frac{\pi^4}8 .
\]
Quantitatively, let $L=\log N\ge16$ and put
\[
\begin{gathered}
 J=\lfloor N/2\rfloor,\qquad k_0=\lceil L^2\rceil,\qquad
 \alpha_J=\frac{2H_J}\pi+1,\\
 \sigma_N=\frac{\sqrt{2\alpha_J^2+6}}{\pi\sqrt{1-2/L^2}},\qquad
 \tau_N=160+2\sqrt{LH_{k_0}}+30\sqrt{L+5}.
\end{gathered}
\]
Then every real or complex $p\in\Pn_N$ satisfies
\begin{equation}\label{eq:sharp-finite}
 \Bigl(\int_{-1}^1|p|^2\Bigr)^{1/2}\le(\sigma_N+\tau_N)\,
 Q_N(|p|^2)^{1/2},
\end{equation}
where $\sigma_N/L\to2\sqrt2/\pi^2$ and $\tau_N/L\to0$.
\end{theorem}

\begin{proof}[Proof of Theorem~\ref{thm:sharp-min}, given \eqref{eq:sharp-finite}]
Inequality \eqref{eq:sharp-finite} gives
\[
 L^2\lambda_-(N,N)\ge\Bigl(\frac{\sigma_N}L+\frac{\tau_N}L\Bigr)^{-2}.
\]
Since $\log(J+1)\le H_J\le1+\log J$, we have $\alpha_J/L\to2/\pi$ and
hence $\sigma_N/L\to2\sqrt2/\pi^2$. Also $H_{k_0}=O(\log L)$, so
$\tau_N/L\to0$. Therefore
$\liminf L^2\lambda_-(N,N)\ge(2\sqrt2/\pi^2)^{-2}=\pi^4/8$.
Restriction gives $\lambda_-(N,N-1)\ge\lambda_-(N,N)$, and
Proposition~\ref{prop:sharp-trial} bounds the upper limit of the former.
\end{proof}

In the language of Section~\ref{sec:context}, Theorem~\ref{thm:sharp-min}
is a sharp converse Marcinkiewicz--Zygmund statement.

% Source: PD-T?mz-paper-claude-01 (d); topics/padua-mz-paper-20261113/theory/cc-defect-condition-corollary.md
\begin{corollary}[Interpolation from weighted nodal data]\label{cor:interp-norm}
Let $\mathcal I_N$ map nodal data $u\in\R^{N+1}$, normed by
$\norm u_w=(\sum_kw_ku_k^2)^{1/2}$, to its interpolant in $\Pn_N$, normed in
$L^2(-1,1)$. Then $\norm{\mathcal I_N}=\lambda_-(N,N)^{-1/2}$ and
\[
 \lim_{N\to\infty}\frac{\norm{\mathcal I_N}}{\log N}=\frac{2\sqrt2}{\pi^2}.
\]
\end{corollary}

\begin{proof}
Interpolation is a bijection, and $\norm u_w^2=Q_N(p^2)$ for
$p=\mathcal I_Nu$, so
$\norm{\mathcal I_N}^2=\sup_{0\ne p\in\Pn_N}\int p^2/Q_N(p^2)=\lambda_-(N,N)^{-1}$.
Theorem~\ref{thm:sharp-min} gives the limit.
\end{proof}

The proofs of Proposition~\ref{prop:sharp-trial} and of
\eqref{eq:sharp-finite} are in Appendix~\ref{app:sharp}. They rest on
two estimates for finitely supported sequences $g=(g_k)_{k\ge1}$,
measured in the harmonic norm
$\norm g_h^2=\sum_{k\ge1}|g_k|^2/k$. The first controls the signed
part of the endpoint kernel without any regularity of $g$; the second is
a discrete version of Lemma~\ref{lem:volterra} with the oscillation
$\sin^2(\pi t)$ averaged uniformly.

\begin{lemma}[Signed remainder]\label{lem:signed-remainder}% IDs: PD-T544
For noninteger $t>0$ put
\[
 \Phi_k(t)=\frac{t^2}{k(t^2-k^2)}-\frac{\boldsymbol1_{k<t}}k,\qquad
 (\mathcal Eg)(t)=\sin(\pi t)\sum_{k\ge1}g_k\Phi_k(t),
\]
extended continuously to integer $t$. Then
\[
 \int_0^\infty|(\mathcal Eg)(t)|^2\frac{\dd t}t\le1600\,\norm g_h^2 .
\]
\end{lemma}

\begin{lemma}[Discrete primitive]\label{lem:discrete-primitive}% IDs: PD-T545
Let $T\ge1$, $J=\lfloor T\rfloor$, $\alpha_J=2H_J/\pi+1$, and let $g$ be
supported in $\{1,\dots,J\}$. With $S(t)=\sum_{k<t}g_k/k$,
\[
 \int_0^T\sin^2(\pi t)\,|S(t)|^2\,\frac{\dd t}t
 \le\Bigl(\frac{\alpha_J^2}2+\frac32\Bigr)\norm g_h^2 .
\]
\end{lemma}

Neither lemma uses smoothness, signs or cancellation of the input. The
first is where signs are retained: the pole part of $\Phi_k$ is summed
by an orthogonality argument, not in absolute value; an absolute
estimate of the cardinal functions loses a logarithm. The second
supplies the leading coefficient: $\alpha_J\sim(2/\pi)\log T$ is the
rescaled norm of $V$, and the factor $1/2$ is the mean of $\sin^2$.

\section{The endpoint maximum eigenvalue}\label{sec:edge-max}

The constant polynomial shows $\lambda_+(N,N-1)\ge1$, and
Theorem~\ref{thm:five-thirds} gives $\lambda_+(N,N-1)\le5/3$. The upper
eigenvalue is again decided at the endpoints, now on the microscopic
scale $\theta=\xi/N$ with $\xi$ bounded. This section identifies its limit
as the norm of an explicit sampling operator.

\paragraph{The endpoint model}
For $g\in L^2(0,1)$ put
\[
 \widehat g(\xi)=\int_0^1g(s)\cos(\xi s)\dd s,\qquad
 \mathsf M(g)=\int_0^\infty\xi\,|\widehat g(\xi)|^2\dd\xi\in[0,\infty],
\]
and, with $\operatorname{Si}(x)=\int_0^x\sin v\,v^{-1}\dd v$,
\[
 \omega_j=2\bigl[(-1)^j+\pi j\operatorname{Si}(\pi j)\bigr]\quad(j\ge1),\qquad
 \mathsf D(g)=|\widehat g(0)|^2+\sum_{j\ge1}\omega_j|\widehat g(\pi j)|^2 .
\]
Let $\mathcal V_0$ be the span of the functions
$\zeta_j(s)=2[\cos(\pi js)-(-1)^j]$, $j\ge1$. Their cosine transforms are
\begin{equation}\label{eq:chi}
 \chi_j(\xi):=\widehat{\zeta_j}(\xi)
 =\frac{2(-1)^j(\pi j)^2\sin\xi}{\xi(\xi^2-(\pi j)^2)},
\end{equation}
with removable singularities filled in; they satisfy
$\chi_j(0)=-2(-1)^j$, $\chi_j(\pi k)=\delta_{jk}$ for $k\ge1$, and
$\chi_j(\xi)=O_j(\xi^{-3})$. Thus for $g=\sum_{j\le J}u_j\zeta_j$,
\[
 \mathsf D(g)=4\Bigl(\sum_{j=1}^J(-1)^ju_j\Bigr)^2+\sum_{j=1}^J\omega_ju_j^2,
 \qquad
 \mathsf M(g)=\int_0^\infty\xi\Bigl(\sum_{j=1}^Ju_j\chi_j(\xi)\Bigr)^2\dd\xi .
\]
Define
\begin{equation}\label{eq:Lambda-edge}
 \Lambda_{\mathrm{edge}}
 =\sup_{0\ne g\in\mathcal V_0}\frac{\mathsf D(g)}{\mathsf M(g)} .
\end{equation}
The numerator is the limit of the rescaled CC energy and the denominator
that of the rescaled continuous mass of the degree-$(N-1)$ polynomials
\begin{equation}\label{eq:pNj}
 p_{N,j}=\ell_j-2(-1)^j\ell_0\qquad(1\le j<N),
\end{equation}
where $\ell_k\in\Pn_N$ are the cardinal polynomials. Their two nonzero
nodal values cancel the coefficient of $T_N$, so $p_{N,j}\in\Pn_{N-1}$.
(Strictly alternating nodal values, by contrast, have a nonzero $T_N$
coefficient and are not admissible.) The weights $\omega_j$ are the limits
of $N^2w_j$, and $\chi_j(\xi)$ is the limit of $p_{N,j}(\cos(\xi/N))$.

\begin{proposition}[Fixed-trial limits]\label{prop:fixed-trial}% IDs: PD-T529 (weakened)
For every $J\ge1$ and $u\in\R^J$, with $g=\sum_ju_j\zeta_j$ and
$p=\sum_ju_jp_{N,j}$,
\[
 N^2\int_{-1}^1p^2\longrightarrow\mathsf M(g),\qquad
 N^2Q_N(p^2)\longrightarrow\mathsf D(g)\qquad(N\to\infty).
\]
The form $\mathsf M$ is positive definite on $\mathcal V_0$,
$1\le\Lambda_{\mathrm{edge}}\le5/3$, and
$\liminf_{N\to\infty}\lambda_+(N,N-1)\ge\Lambda_{\mathrm{edge}}$ through
all integers $N$.
\end{proposition}

\begin{theorem}[The endpoint upper limit]\label{thm:edge-limit}% IDs: PD-T530
Through all integers $N$,
\[
 \lim_{N\to\infty}\lambda_+(N,N-1)=\Lambda_{\mathrm{edge}},\qquad
 1\le\Lambda_{\mathrm{edge}}\le\frac32 .
\]
\end{theorem}

The convergence is qualitative: the proof (Appendix~\ref{app:edge}) uses
weak compactness and gives no rate. Its idea is as follows. In the
decomposition $R=M-Y+Z$ of Theorem~\ref{thm:five-thirds}, the rescaled
transition matrix $Z$ converges to a compact integral operator, while the
reflected matrix $Y$ is positive semidefinite and only lower semicontinuous
under weak convergence. Any normalized coefficient sequence produces two
weak endpoint profiles, one for each endpoint, and these profiles are
compared with the model through the continuum identity
$\mathsf D=\mathsf M+\tfrac12(\mathsf Z-\mathsf Y)$ on the closure of
$\mathcal V_0$.

% Source: PD-T547; topics/padua-mz-sharp-constants-20261106/theory/P4-certified-lower.md
\begin{proposition}[A certified lower bound]\label{prop:edge-certified}% IDs: PD-T547
$\Lambda_{\mathrm{edge}}\ge1.0044369$.
\end{proposition}

The proof (Appendix~\ref{app:certificate}) is computer-assisted: it evaluates,
in outward rational interval arithmetic, the quotient
$\mathsf D(g)/\mathsf M(g)$ for one explicit $g\in\mathcal V_0$ with $512$
coefficients, using closed forms of $\mathsf M$ on $\mathcal V_0$. It is a
lower bound only. Floating-point Galerkin computations with the same basis,
and the eigenvalues in Table~\ref{tab:cc}, suggest that
$\Lambda_{\mathrm{edge}}$ agrees with $1.0044369$ to several further digits,
but these are measurements and no digit beyond the certified bound is
claimed. Determining the decimal value of $\Lambda_{\mathrm{edge}}$ is
left open (Section~\ref{sec:numerics}).

\section{Consequences at the endpoint}\label{sec:consequences}

Write $G_N=G_{N,N-1}$ and $\eta_N=\norm{I-G_N}_2$. Since the extreme
eigenvalues determine the distance from the identity,
\begin{equation}\label{eq:defect-identity}
 1-\eta_N=\min\{\lambda_-(N,N-1),\,2-\lambda_+(N,N-1)\}.
\end{equation}

% Sources: PD-T527; PD-T?mz-paper-claude-01 (a)(b); topics/padua-mz-paper-20261113/theory/cc-defect-condition-corollary.md
\begin{corollary}[Defect and condition number at the endpoint]\label{cor:endpoint}% IDs: PD-T527
For every $N\ge2$, with $K_N$ from \eqref{eq:KN},
\[
 \min\{K_N^{-1},1/3\}\le1-\eta_N\le\min\{1,128000/(\log N)^2\}.
\]
For every $N\ge57$, $1-\eta_N=\lambda_-(N,N-1)$. Consequently
\[
 \lim_{N\to\infty}(\log N)^2(1-\eta_N)=\frac{\pi^4}8,\qquad
 \lim_{N\to\infty}\frac{\kappa_2(G_N)}{(\log N)^2}=\frac{8\Lambda_{\mathrm{edge}}}{\pi^4}.
\]
The last limit lies in $[8/\pi^4,12/\pi^4]$, and in
$[8\cdot1.0044369/\pi^4,\,12/\pi^4]\subset[0.08249,0.12320]$ by the
computer-assisted Proposition~\ref{prop:edge-certified}.
\end{corollary}

\begin{proof}
By Theorem~\ref{thm:five-thirds}, $2-\lambda_+(N,N-1)\ge1/3$. Together with
Theorems~\ref{thm:cardinal} and~\ref{thm:harmonic}, the constant polynomial
($\lambda_-\le1$) and \eqref{eq:defect-identity}, this gives the two-sided
bound. If $\lambda_-(N,N-1)\le1/3$, \eqref{eq:defect-identity} reduces to
$1-\eta_N=\lambda_-(N,N-1)$; Lemma~\ref{lem:U-exact} gives this for
$N\ge57$. The first limit is then Theorem~\ref{thm:sharp-min}. Since
$\lambda_->0$, $\kappa_2(G_N)=\lambda_+/\lambda_-$, and the second limit follows
from Theorems~\ref{thm:sharp-min} and~\ref{thm:edge-limit}. The intervals use
$1\le\Lambda_{\mathrm{edge}}\le3/2$ and Proposition~\ref{prop:edge-certified}.
\end{proof}

\begin{remark}[What remains open at the endpoint]\label{rem:rate}
The endpoint constants $\pi^4/8$ and $\Lambda_{\mathrm{edge}}$ are determined
as limits, but the proofs give no rate of convergence that is usable at
computable $N$: the finite bound \eqref{eq:sharp-finite} is coarse, and
Theorem~\ref{thm:edge-limit} is qualitative. The decimal value of
$\Lambda_{\mathrm{edge}}$ is known only through the one-sided
Proposition~\ref{prop:edge-certified} and the measurements of
Section~\ref{sec:numerics}.
\end{remark}

\section{Explicit bounds at each fixed degree ratio}\label{sec:ratio}

For $n\le N/2$, exactness gives $\lambda_\pm(N,n)=1$; the classical
algebraic Marcinkiewicz inequality of Mastroianni--Totik \cite{LIT-051}
yields nothing beyond this in its documented form
(Section~\ref{sec:context}). The point of this section is the range
$N/2<n<N$: explicit constants for every ratio below one, with an upper
constant below two.

\begin{theorem}[Filtered lower estimate]\label{thm:ratio-lower}% IDs: PD-T023
Put $C_0=2(1+\pi)^2$ and
$C_*=3C_0^2(1+2/\pi)=12(1+\pi)^4(1+2/\pi)$.
For $N\ge2$, $0\le n<N$, and $p\in\Pn_n$,
\begin{equation}\label{eq:fixedlower}
 \int_{-1}^1p^2\le\frac{C_*}{(1-n/N)^2}Q_N(p^2).
\end{equation}
\end{theorem}
\begin{proof}
On the circle $\R/(2\pi\mathbb Z)$ let $d(t,s)\in[0,\pi]$ denote
shorter angular distance, $t_j=j\pi/N$ for $0\le j<2N$,
$q(t)=p(\cos t)$, $\nu(t)=|\sin t|+1/N$, and $\epsilon=1-n/N$.
Integrals on this circle use ordinary $\dd t$; discrete sums use counting
measure unless a factor is written.

For $m\ge1$, the finite Fej\'er identity is
\[
 F_m(t)=\sum_{|k|<m}(1-|k|/m)e^{ikt}
       =\frac1m\left|\sum_{h=0}^{m-1}e^{iht}\right|^2.
\]
Counting pairs with difference $k$ proves the equality. Define
\begin{equation}\label{eq:filter}
 K(t)=\frac{NF_N(t)-nF_n(t)}{N-n},
\end{equation}
where $nF_n=0$ when $n=0$. Its coefficient at frequency $k$ is
$[\max(N-|k|,0)-\max(n-|k|,0)]/(N-n)$.
Thus $K$ has degree $N-1$ and coefficient one for every $|k|\le n$,
including the boundary $|k|=n$. Finite Fourier expansion gives
\begin{equation}\label{eq:reconstruction}
 q(t)=\frac1{2N}\sum_{j=0}^{2N-1}K(t-t_j)q(t_j).
\end{equation}
Indeed the grid average of $e^{i(l-k)t_j}$ vanishes unless $l-k$ is
a multiple of $2N$. Since $|l-k|\le n+N-1<2N$, only $l=k$ remains;
its multiplier is one. No continuous sampling theorem is assumed.

Let $d=d(t,0)$. The geometric-sum formula gives
$mF_m(t)\le m^2$ and $mF_m(t)\le\pi^2/d^2$ because
$\sin(d/2)\ge d/\pi$. The triangle inequality in \eqref{eq:filter}
therefore gives
\begin{equation}\label{eq:filter-decay}
 |K(t)|\le\frac{2N}{\epsilon}\min\{1,\pi^2/(Nd)^2\}
       \le\frac{C_0N}{\epsilon(1+Nd)^2}.
\end{equation}
The last scalar bound follows by splitting $Nd\le\pi$ and $Nd\ge\pi$;
the value at zero follows by continuity.
The function $\nu$ is one-Lipschitz for circle distance and at least $1/N$,
so $\nu(t)/\nu(t_j)\le1+Nd(t,t_j)$. Set
\[
 z_j=\sqrt{\nu(t_j)}q(t_j),\qquad
 H(t,j)=\frac1{2N}K(t-t_j)\sqrt{\frac{\nu(t)}{\nu(t_j)}}.
\]
Then $\sqrt{\nu(t)}q(t)=\sum_jH(t,j)z_j$ and
\begin{equation}\label{eq:Hdecay}
 |H(t,j)|\le\frac{C_0}{2\epsilon}(1+Nd(t,t_j))^{-3/2}.
\end{equation}
Each half-open shell $[k\pi,(k+1)\pi)$ of the scaled distance $Nd$
contains at most two nodes, one in each circle direction. This is because
each corresponding half-open arc has the grid spacing $\pi/N$ as its length.
Hence, using a decreasing-sum integral comparison,
\begin{align}
 \sup_t\sum_j|H(t,j)|
 &\le\frac{C_0}{\epsilon}\sum_{k\ge0}(1+k\pi)^{-3/2}
 \le\frac{C_0}{\epsilon}(1+2/\pi)=:R,\label{eq:rowH}\\
 \sup_j\int_0^{2\pi}|H(t,j)|\dd t
 &\le\frac{C_0}{2\epsilon}\frac2N
                 \int_0^\infty(1+u)^{-3/2}\dd u
 =\frac{2C_0}{\epsilon N}=:S.\label{eq:colH}
\end{align}
At an antipode the node count can only decrease. Cauchy--Schwarz gives
$|\sum_jH(t,j)z_j|^2\le(\sum_j|H(t,j)|)
\sum_j|H(t,j)||z_j|^2$.
Integrating and using \eqref{eq:rowH}--\eqref{eq:colH} proves
\begin{equation}\label{eq:circlebound}
 \int_0^{2\pi}\nu(t)|q(t)|^2\dd t
 \le\frac{2C_0^2(1+2/\pi)}{\epsilon^2N}
          \sum_{j=0}^{2N-1}\nu(t_j)|q(t_j)|^2.
\end{equation}

For each interior half-circle node, $\sin t_j\ge2/N$ and
$\nu(t_j)\le(3/2)\sin t_j$. Lemma~\ref{lem:weight-comparison} and
the endpoint formulas imply at every half-circle node
$w_j\ge2\nu(t_j)/(3N)$. Reflection then gives
\[
 \sum_{j=0}^{2N-1}\nu(t_j)|q(t_j)|^2
 \le2\sum_{j=0}^N\nu(t_j)|p(x_j)|^2\le3NQ_N(p^2).
\]
The first inequality overcounts the two endpoints, which is harmless.
Finally $2\int_{-1}^1p^2=\int_0^{2\pi}|\sin t|q(t)^2\dd t
\le\int_0^{2\pi}\nu(t)q(t)^2\dd t$.
Substitution in \eqref{eq:circlebound} proves \eqref{eq:fixedlower}.
\end{proof}

\begin{corollary}[Explicit strong constants]\label{cor:strong}% IDs: PD-T024
For $0\le\theta<1$, set
\[
 A_\theta=\frac{(1-\theta)^2}{C_*},\qquad
 \rho_\theta=\max\{2\theta-1,0\}^2,\qquad
 \eta_\theta=\max\{1-A_\theta,\rho_\theta\}<1.
\]
For every integer $0\le n\le\theta N$, the spectrum of $G_{N,n}$ lies
in $[A_\theta,1+\rho_\theta]$; consequently
\[
 (1-\eta_\theta)\int p^2\le Q_N(p^2)
                    \le(1+\eta_\theta)\int p^2\qquad(p\in\Pn_n).
\]
\end{corollary}
\begin{proof}
The two preceding theorems give the spectral interval by monotonicity in
$n/N$. Since $C_*>1$, $0<A_\theta<1$ and $0\le\rho_\theta<1$.
The definition of $\eta_\theta$ gives both inequalities.
\end{proof}

The filtered estimate degenerates like $N^{-2}$ at $n=N-1$ and does not
replace the sharper endpoint theorem. Conversely the endpoint cardinal
argument does not supply a constant independent of $N$ at a fixed ratio.
Both proofs are included because the two regimes need different estimates.

\section{The three-quarter assumption and hyperinterpolation}\label{sec:hyper}

An--Wu's inspected arXiv version assumes a relative squared-norm MZ error
$\eta<1$ in its main theorem. Its Section~4 assumes $\eta=3/4$ for the
examples, and Corollary~4.1 retains the theorem's conditions for an
$m$-point CC rule with $n+2\le m\le2n+1$, obtaining an error bound
$3\sqrt V E_{m-1-n}(f)$ \cite{LIT-046}.
The following finite calculation specifies an excluded case for that
assumption; it does not refute the conditional implication.

\begin{proposition}[An exact endpoint witness]\label{prop:counterexample}% IDs: PD-T019, PD-N004
$\lambda_-(512,511)<1/4$; in particular $\norm{I-G_{512,511}}_2>3/4$.
This case has $513$ CC points, so An--Wu's point-count notation is
$m=513$, $n=511$.
\end{proposition}
\begin{proof}
We use Lemma~\ref{lem:U-exact}. For a short exact certificate at $N=512$, direct rational addition gives
$S_8>201/100$. Monotonicity of $1/(2t-1)$ gives
\[
 S_{512}\ge S_8+\int_9^{513}\frac{\dd t}{2t-1}
          =S_8+\tfrac12\log(1025/17).
\]
The factorial series for $e$ gives
$e\le1+1+1/2+1/6+(1/24)/(1-1/5)=87/32<11/4$.
The rational inequality $(11/4)^4<1025/17$ implies
$\log(1025/17)>4$, so $S_{512}>401/100$.
Since $4\cdot512^2/(512^2-1)<401/100$, \eqref{eq:Uexact} yields
\[
 \frac{Q_{512}(U_{511}^2)}{\int U_{511}^2}
 =\frac{512^2}{(512^2-1)S_{512}}<\frac14.
\]
Thus $\lambda_-(512,511)<1/4$ and the defect exceeds $3/4$.
\end{proof}

Lemma~\ref{lem:U-exact} alone already gives $q_{512}<1/3$; the certificate
above is the sharper value needed for the threshold $3/4$. Analogous borderline mechanisms appear in classical
converse Marcinkiewicz--Zygmund characterizations \cite{LIT-293,LIT-294},
and a full-space, subsequence version of $\lambda_-(N,N)\to0$ at the CC
nodes follows from \cite{LIT-299} (Section~\ref{sec:context}).
Lemma~\ref{lem:U-exact} supplies the exact CC evaluation, valid for every
$N$ and on $\Pn_{N-1}$. The finite certificate uses no floating spectrum.
The published An--Wu Corollary~4.1 coincides with the inspected arXiv
formulation: it assumes $\eta=3/4$ for $m$-point CC with $n+2\le m\le2n+1$
\cite{LIT-046}, and $m=513$, $n=511$ lies in that range. No correction
was found in a documented search; this is not proof of absence.

\begin{proposition}[Upper-Gram hyperinterpolation estimate]\label{prop:hyper}% IDs: PD-T025
Let $\mu$ be a finite positive measure of mass $V>0$ on a compact domain.
Let $Q$ have positive weights of total mass $V$, and let
$P_1,\ldots,P_d$ be an $L^2(\mu)$-orthonormal basis of $\Pn_n$.
Assume $Q(p^2)\le B\norm p_2^2$ on $\Pn_n$, and define
\[
 \mathcal L_nf=\sum_{i=1}^d Q(fP_i)P_i.
\]
Then $\norm{\mathcal L_nf}_2\le\sqrt{VB}\norm f_\infty$.
If $0\le k\le n$ and $Q$ is exact on all products of a member of
$\Pn_k$ with a member of $\Pn_n$, then
\begin{equation}\label{eq:hypererror}
 \norm{f-\mathcal L_nf}_2\le\sqrt V(1+\sqrt B)E_k(f),\qquad
 E_k(f)=\inf_{r\in\Pn_k}\norm{f-r}_\infty.
\end{equation}
\end{proposition}
\begin{proof}
For $c_i=Q(fP_i)$ and any Euclidean unit vector $a$, put
$p_a=\sum_i a_iP_i$, so $\norm{p_a}_2=1$.
Weighted Cauchy--Schwarz gives
$|a\cdot c|=|Q(fp_a)|\le Q(f^2)^{1/2}Q(p_a^2)^{1/2}
\le\sqrt{VB}\norm f_\infty$.
Euclidean duality and orthonormality identify
$\norm{\mathcal L_nf}_2=\norm c_2$, proving the first bound.
For $r\in\Pn_k$, product exactness gives
$Q(rP_i)=\int rP_i\dd\mu$, and since $k\le n$,
$\mathcal L_nr=r$. Hence
\[
 \norm{f-\mathcal L_nf}_2
 \le\norm{f-r}_2+\norm{\mathcal L_n(f-r)}_2
 \le\sqrt V(1+\sqrt B)\norm{f-r}_\infty.
\]
Taking the infimum proves \eqref{eq:hypererror}.
\end{proof}

For CC, $V=2$, $0\le n<N$, and one may take
% Source: PD-T025 CC application; PD-T?mz-paper-claude-01 (c)
\begin{equation}\label{eq:Bcc}
 B_{N,n}=\min\Bigl\{1+\rho_{N,n},\,\frac53\Bigr\},\qquad
 \rho_{N,n}=\max\{2n/N-1,0\}^2,
\end{equation}
by Theorems~\ref{thm:ratio-upper} and~\ref{thm:five-thirds}; the bound $5/3$
is the smaller one exactly when $n/N>(1+\sqrt{2/3})/2=0.9082\ldots$.
Thus for every $0\le k\le\min(n,d_N-n)$,
\begin{equation}\label{eq:CCerror}
 \norm{f-\mathcal L_nf}_{L^2(-1,1)}
 \le\sqrt2(1+\sqrt{B_{N,n}})E_k(f)
 \le\Bigl(1+\sqrt{5/3}\Bigr)\sqrt2\,E_k(f),
\end{equation}
and $1+\sqrt{5/3}<2.2910$.
Using only exactness through $N$ permits $k\le\min(n,N-n)$;
even-$N$ parity can improve that by one when $k\le n$ allows it.
For the $513$-point, degree-$511$ example, $k=1$ already gives the
uniform factor $(1+\sqrt{5/3})\sqrt V$ in place of the assumed
$3\sqrt V$ factor; parity also permits $k=2$ here.
This estimate concerns the defined quadrature hyperinterpolant. An
inverse-Gram least-squares operator is a different construction; growth
of $\lambda_-^{-1/2}$ does not describe the norm of $\mathcal L_n$.
Neither the general mechanism nor the validity of An--Wu's conditional
theorem is challenged by this CC application.

\section{Proofs for the sharp endpoint constant}\label{app:sharp}

Throughout this appendix $N\ge2$, $L=\log N$, $\theta_k=k\pi/N$,
$d_k=\min(k,N-k)$, and $c_0=c_N=2$, $c_k=1$ for $0<k<N$ (the
reciprocal endpoint factors of the cardinal formula in the proof of
Theorem~\ref{thm:cardinal}). For nodal data $u=(u_0,\dots,u_N)$ let
$\mathcal I_Nu\in\Pn_N$ be its interpolant, $a_k=(-1)^ku_k$, and
$\norm u_w^2=\sum_{k=0}^Nw_k|u_k|^2$, so that
$\norm u_w^2=Q_N(|\mathcal I_Nu|^2)$. All $L^2$ norms of functions on
$[-1,1]$ are with respect to $\dd x$; on $(0,1)$ or $(0,\infty)$ the
measure is written explicitly. The harmonic norm of a finitely supported
sequence $g=(g_k)_{k\ge1}$ is $\norm g_h^2=\sum_k|g_k|^2/k$.

\subsection{\texorpdfstring{Proof of Lemma~\ref{lem:signed-remainder}}{Proof of the signed remainder lemma}}

Put $\operatorname{sinc}(x)=\sin(\pi x)/(\pi x)$, $\operatorname{sinc}(0)=1$.
For integers $k,l$,
\begin{equation}\label{eq:sinc-orth}
 \int_{\R}\operatorname{sinc}(t-k)\operatorname{sinc}(t-l)\dd t=\boldsymbol1_{k=l}.
\end{equation}
For $k=l$ this is $\int_{\R}\sin^2(\pi t)/(\pi t)^2\dd t=1$, which follows
from $\int_0^\infty\sin(bu)\,u^{-1}\dd u=\pi/2$ ($b>0$) after one integration
by parts. For $k\ne l$ the integer shifts give
$\operatorname{sinc}(t-k)\operatorname{sinc}(t-l)=(-1)^{k+l}\sin^2(\pi t)/[\pi^2(t-k)(t-l)]$,
and partial fractions reduce the integral to differences of
$\int_{-R}^R\sin^2(\pi t)(t-k)^{-1}\dd t$, which tend to zero as
$R\to\infty$: after the shift $t\mapsto t+k$ the integrand is odd on
$[-R,R]$ and the two end intervals of length $|k|$ contribute $O_k(1/R)$.
Since the product is absolutely integrable, its integral is the limit,
zero. Consequently, for every finite sequence $(h_k)$,
\begin{equation}\label{eq:sinc-isometry}
 \int_{\R}\Bigl|\sum_kh_k\operatorname{sinc}(t-k)\Bigr|^2\dd t=\sum_k|h_k|^2 .
\end{equation}

For $j\in\mathbb Z$ let $\delta=2^j$, $I_j=[\delta,2\delta)$ and
$\mathcal J_j=\{k\ge1:\delta/2<k<4\delta\}$; each $k\ge1$ lies in at most
four sets $\mathcal J_j$. For $t\in I_j$ split $\sum_kg_k\Phi_k(t)$ into
$k\in\mathcal J_j$ and $k\notin\mathcal J_j$. Since
$t^2/[k(t^2-k^2)]=1/k+\tfrac12[1/(t-k)-1/(t+k)]$, one has exactly
\begin{equation}\label{eq:Phi-split}
 \Phi_k(t)=\frac1{2(t-k)}+\Psi_k(t),\qquad
 \Psi_k(t)=-\frac1{2(t+k)}+\frac{\boldsymbol1_{k>t}}k .
\end{equation}

\emph{The pole part.} As $\sin(\pi t)=(-1)^k\sin(\pi(t-k))$,
\[
 \sin(\pi t)\sum_{k\in\mathcal J_j}\frac{g_k}{2(t-k)}
 =\frac\pi2\sum_{k\in\mathcal J_j}(-1)^kg_k\operatorname{sinc}(t-k).
\]
Since $t\ge\delta$ on $I_j$ and $k<4\delta$ on $\mathcal J_j$,
\eqref{eq:sinc-isometry} bounds its $L^2(I_j,\dd t/t)$ norm squared by
$(\pi^2/4\delta)\sum_{\mathcal J_j}|g_k|^2\le\pi^2\sum_{\mathcal J_j}|g_k|^2/k$.
Summing over $j$, the pole part has norm at most $2\pi<7$ times
$\norm g_h$. This is the only place where the signs of the cardinal
kernel matter, and they are kept.

\emph{The comparable scales.} For $k\in\mathcal J_j$ and $t\in I_j$ we
have $k>t/4$, so $|\Psi_k(t)|\le1/(2t)+4/t=9/(2t)$. A nonempty
$\mathcal J_j$ forces $\delta\ge1/2$, whence
$\sum_{k\in\mathcal J_j}k\le8\delta^2+2\delta\le12\delta^2\le12t^2$.
Cauchy--Schwarz gives
\[
 \Bigl|\sum_{k\in\mathcal J_j}g_k\Psi_k(t)\Bigr|^2
 \le\Bigl(\sum_{\mathcal J_j}\frac{|g_k|^2}k\Bigr)
 \Bigl(\sum_{\mathcal J_j}k|\Psi_k(t)|^2\Bigr)
 \le243\sum_{\mathcal J_j}\frac{|g_k|^2}k .
\]
Integrating over $I_j$ against $\dd t/t$ (mass $\log2$) and summing with
the fourfold overlap bounds this part by $(972\log2)^{1/2}\norm g_h<32\norm g_h$.

\emph{The separated scales.} For $k\notin\mathcal J_j$ either
$k\le t/2$ or $k\ge2t$. The kernel $\mathcal K(t,k)=k\Phi_k(t)$ equals
$k^2/(t^2-k^2)$ for $k<t$ and $t^2/(t^2-k^2)$ for $k>t$, so
\[
 |\mathcal K(t,k)|\le\tfrac43(k/t)^2\ (k\le t/2),\qquad
 |\mathcal K(t,k)|\le\tfrac43(t/k)^2\ (k\ge2t).
\]
Its weighted row sums satisfy $\sum_{k\le t/2}|\mathcal K(t,k)|/k
\le(4/3t^2)\sum_{k\le t/2}k\le1/3$ (the sum is empty unless $t\ge2$),
and, with $n=\lceil2t\rceil$,
$\sum_{k\ge2t}|\mathcal K(t,k)|/k\le\tfrac43t^2(n^{-3}+\tfrac12n^{-2})
\le\tfrac1{6t}+\tfrac16\le\tfrac12$ for $t\ge1/2$, while for $t<1/2$
it is at most $\tfrac43t^2\sum_kk^{-3}<1/2$. Its weighted column
integrals satisfy
$\int_0^\infty|\mathcal K(t,k)|\,\dd t/t\le\tfrac43
(\int_0^{k/2}(t/k)^2\dd t/t+\int_{2k}^\infty(k/t)^2\dd t/t)=\tfrac13$.
By the Schur test in the form
$|\sum_k\mathcal K g_k/k|^2\le(\sum_k|\mathcal K|/k)(\sum_k|\mathcal K||g_k|^2/k)$,
integrated in $\dd t/t$, this part has norm at most one.

The three parts have norms below $7$, $32$ and $1$; multiplication by
$\sin(\pi t)$ does not increase them. Hence the norm of $\mathcal E$ is at
most $40$, which is the lemma. \qed

\subsection{\texorpdfstring{Proof of Lemma~\ref{lem:discrete-primitive}}{Proof of the discrete primitive lemma}}

Put $H_0=0$ and let $f=g_m$ on $(H_{m-1},H_m]$, $1\le m\le J$, so that
$\norm f_{L^2(0,H_J)}^2=\norm g_h^2$. Its primitive
$F(x)=\int_0^xf$ satisfies $F(H_m)=S_m:=\sum_{k\le m}g_k/k$. Let $Z$
equal $S_m$ on $(H_{m-1},H_m]$. On that interval
$Z-F=g_m(1/m-(x-H_{m-1}))$, so $\int|Z-F|^2=|g_m|^2/(3m^3)$ and
$\norm{Z-F}\le\norm g_h$. Lemma~\ref{lem:volterra}, rescaled to an
interval of length $H_J$, gives $\norm F\le(2H_J/\pi)\norm g_h$. Hence
\begin{equation}\label{eq:Sm-sum}
 \sum_{m=1}^J\frac{|S_m|^2}m=\norm Z^2\le\alpha_J^2\norm g_h^2 .
\end{equation}
For integers $m\ge1$, since $\int_m^{m+1}(\sin^2(\pi t)-\tfrac12)\dd t=0$,
\[
 \int_m^{m+1}\sin^2(\pi t)\frac{\dd t}t
 =\frac12\log\Bigl(1+\frac1m\Bigr)
 +\int_m^{m+1}\Bigl(\sin^2(\pi t)-\frac12\Bigr)\Bigl(\frac1t-\frac1m\Bigr)\dd t
 \le\frac1{2m}+\frac1{4m^2}.
\]
Cauchy--Schwarz and $H_m\le1+\log m\le2\sqrt m$ give
$|S_m|^2\le H_m\norm g_h^2\le2\sqrt m\,\norm g_h^2$, so
$\sum_{m\ge1}|S_m|^2/m^2\le2\zeta(3/2)\norm g_h^2\le6\norm g_h^2$.
On $(m,m+1)$ one has $S(t)=S_m$, and $S=0$ on $(0,1)$. Enlarging the
last partial cell to a full cell and summing,
\[
 \int_0^T\sin^2(\pi t)|S(t)|^2\frac{\dd t}t
 \le\sum_{m=1}^J|S_m|^2\Bigl(\frac1{2m}+\frac1{4m^2}\Bigr)
 \le\Bigl(\frac{\alpha_J^2}2+\frac32\Bigr)\norm g_h^2. \qquad\qed
\]

\subsection{The endpoint representation of the interpolant}

\begin{lemma}\label{lem:endpoint-rep}
Let $N\ge2$.
\begin{enumerate}
\item[(a)] $w_k\ge(1+d_k)/N^2$ for $0\le k\le N$; for $0<k<N$,
$w_k\ge\sin\theta_k/N$ and $|w_k-(\pi/N)\sin\theta_k|\le3/N^2$. Moreover
$\sum_{k=0}^Nw_k^{-1}\le2N^2(L+1)$.
\item[(b)] For $0\le t\le N/2$ and $\theta=\pi t/N$,
\[
 (\mathcal I_Nu)(\cos\theta)=Y_u(t)+r_u(\theta),\qquad
 Y_u(t)=\frac{2t\sin(\pi t)}\pi\sum_{k=0}^N\frac{a_k}{c_k(t^2-k^2)},
\]
with poles removed by continuity, and
$|r_u(\theta)|\le2\sqrt{2(L+1)}\,\norm u_w$.
\item[(c)] Let $1\le\ell<N/4$ and let $u$ vanish outside
$\{1,\dots,\ell\}\cup\{N-\ell,\dots,N-1\}$. Put $g_k=ka_k/N$ and
$g^\flat_k=ka_{N-k}/N$ for $1\le k\le\ell$ ($g,g^\flat=0$ otherwise), and
let $S$, $S^\flat$ be their partial sums $\sum_{k<t}g_k/k$. Define
$P_u$ on $[-1,1]$ by
\[
 P_u(\cos\theta)=\frac{2N\sin(\pi t)}{\pi t}S(t),\qquad
 P_u(-\cos\theta)=(-1)^N\frac{2N\sin(\pi t)}{\pi t}S^\flat(t),
\]
for $\theta=\pi t/N\in[0,\pi/2]$. Then
$\norm g_h^2+\norm{g^\flat}_h^2\le\norm u_w^2$ and
\[
 \norm{\mathcal I_Nu-P_u}_{L^2(-1,1)}\le\bigl(80+6\sqrt{L+1}\bigr)\norm u_w .
\]
\end{enumerate}
\end{lemma}

\begin{proof}
(a) The first two bounds are Lemma~\ref{lem:weight-comparison}. Its proof
gives $|w_k-(\pi/N)\sin\theta_k|\le2/(N^2-1)$ for even $N$ and $2/N^2$
for odd $N$, both at most $3/N^2$. Each value of $d_k$ occurs at most
twice, so $\sum_kw_k^{-1}\le2N^2H_{\lfloor N/2\rfloor+1}\le2N^2(L+1)$.

(b) By the cardinal formula in the proof of Theorem~\ref{thm:cardinal},
\[
 (\mathcal I_Nu)(\cos\theta)=\frac{\sin(N\theta)}N\sum_{k=0}^N\frac{a_k}{c_k}
 \cdot\frac{-\sin\theta}{\cos\theta-\cos\theta_k} .
\]
With $e(z)=\cot(z/2)-2/z$, extended by $e(0)=0$,
\[
 \frac{-\sin\theta}{\cos\theta-\cos\phi}
 =\frac12\Bigl[\cot\frac{\theta-\phi}2+\cot\frac{\theta+\phi}2\Bigr]
 =\frac{2\theta}{\theta^2-\phi^2}+\frac{e(\theta-\phi)+e(\theta+\phi)}2 .
\]
For $\theta=\pi t/N$ and $\phi=\theta_k$ the first term contributes exactly
$Y_u(t)$. The function $e$ is odd and nonincreasing on $(0,2\pi)$, since
$e'(z)=2/z^2-\tfrac12\csc^2(z/2)\le0$; hence $|e|\le1+4/(3\pi)<2$ on
$[-\pi,3\pi/2]$, which contains $\theta\pm\theta_k$ for $\theta\le\pi/2$.
Thus $|r_u|\le(2/N)\sum_k|u_k|/c_k\le(2/N)(\sum_kw_k^{-1})^{1/2}\norm u_w$,
and (a) gives the bound.

(c) Since $k\le\ell<N/4$, (a) gives $w_k\ge k/N^2$ and
$w_{N-k}\ge k/N^2$, so
$\norm g_h^2+\norm{g^\flat}_h^2=\sum_{k\le\ell}k(|a_k|^2+|a_{N-k}|^2)/N^2
\le\norm u_w^2$. Consider $x\in[0,1]$, i.e.\ $0\le t\le N/2$, and split
$u=u^{\mathrm l}+u^{\mathrm r}$ into its parts near $x=1$ and near $x=-1$.
For the near part, $a_k=Ng_k/k$ and
$t/(t^2-k^2)=t^{-1}\,t^2/(t^2-k^2)$, so by \eqref{eq:Phi-split}
\[
 Y_{u^{\mathrm l}}(t)=\frac{2N\sin(\pi t)}{\pi t}\,S(t)
 +\frac{2N}{\pi t}(\mathcal Eg)(t) .
\]
On $[0,1]$ the measure is $\dd x=\sin\theta\dd\theta\le\pi^2t\dd t/N^2$.
Hence, by Lemma~\ref{lem:signed-remainder}, the second term has
$L^2(0,1)$ norm at most
$(4\int_0^\infty|\mathcal Eg|^2\dd t/t)^{1/2}\le80\norm g_h$.
For the far part, $k\ge N-\ell\ge3N/4$ and $t\le N/2$ give
$k^2-t^2\ge5N^2/16$, so
$|2t/(\pi(t^2-k^2))|\le3t/N^2$ and, by (a),
\[
 |Y_{u^{\mathrm r}}(t)|\le\frac{3t}{N^2}\sum_k|u^{\mathrm r}_k|
 \le\frac{3t}N\sqrt{2(L+1)}\,\norm{u^{\mathrm r}}_w .
\]
Its squared $L^2(0,1)$ norm is at most
$\int_0^{N/2}18(L+1)\norm{u^{\mathrm r}}_w^2\,t^2N^{-2}\cdot\pi^2tN^{-2}\dd t
=\tfrac{9\pi^2}{32}(L+1)\norm{u^{\mathrm r}}_w^2<3(L+1)\norm{u^{\mathrm r}}_w^2$.
The interval $[-1,0]$ is treated by the reflection
$(\mathcal I_Nu)(-x)=(\mathcal I_Nu^R)(x)$, $u^R_k=u_{N-k}$, which preserves
$\norm\cdot_w$ and turns $a_{N-k}$ into $(-1)^N$ times the corresponding
coefficient of $u^R$. On the whole interval the signed terms contribute at most
$80(\norm g_h^2+\norm{g^\flat}_h^2)^{1/2}\le80\norm u_w$, the far parts at most
$\sqrt{3(L+1)}\norm u_w\le2\sqrt{L+1}\norm u_w$, and the remainders of (b)
at most $(2\cdot8(L+1))^{1/2}\norm u_w=4\sqrt{L+1}\norm u_w$.
\end{proof}

\subsection{\texorpdfstring{Proof of Proposition~\ref{prop:sharp-trial}}{Proof of the trial proposition}}

Fix $0<a<b<1$ with $\operatorname{supp}f\subset(a,b)$. The nonzero
values $u_k$, $k<N/2$, have $N^a<k<N^b$; for large $N$ we have
$N^b<\ell:=\lfloor N/L\rfloor<N/4$, so Lemma~\ref{lem:endpoint-rep}(c)
applies. Here $a_k=(-1)^ku_k=(N/k)f(\log k/L)$ and
$a_{N-k}=(-1)^{N-k}(-1)^{N-1}u_k=-a_k$, so $g_k=f(\log k/L)$ and
$g^\flat_k=-g_k$. The top Chebyshev
coefficient of $\mathcal I_Nu$ is $N^{-1}\sum_ka_k/c_k$, which vanishes
because $a_{N-k}=-a_k$; hence $p_N=\mathcal I_Nu\in\Pn_{N-1}$.

\emph{Discrete energy.} By Lemma~\ref{lem:endpoint-rep}(a) and
$\sin z=z(1+O(z^2))$, $w_k=(\pi^2k/N^2)(1+O(N^{2b-2}+N^{-a}))$ uniformly on
the support. Hence
\[
 Q_N(p_N^2)=2\pi^2\sum_{k}\frac{f(\log k/L)^2}k\,(1+o(1))
 =2\pi^2L\norm f^2(1+o(1)),
\]
where the last step compares the sum with
$\int_1^\infty f(\log u/L)^2u^{-1}\dd u=L\norm f^2$; the function
$u\mapsto f(\log u/L)^2/u$ vanishes for $u<N^a$ and has total variation
$O_f(N^{-a})$ on $[N^a,\infty)$.

\emph{Continuous mass.} By the same comparison,
$|S(t)-LF(\log t/L)|\le C_fN^{-a}$ for all $t$, where $F=Vf$ is extended by
$F(b)$ beyond $b$. Fix $0<\delta<1/2$ and let
$I_\delta=[N^\delta,N^{1-\delta}]$. There $\theta\le\pi N^{-\delta}$ and
$\sin\theta\dd\theta\ge(1-\pi^2N^{-2\delta}/6)\pi^2t\dd t/N^2$, so
\begin{align*}
 \int_0^1|P_u|^2\dd x&\ge(1-o(1))\,4\int_{I_\delta}\sin^2(\pi t)|S(t)|^2\frac{\dd t}t\\
 &\ge(1-o(1))\,4L^2\int_{I_\delta}\sin^2(\pi t)F\Bigl(\frac{\log t}L\Bigr)^2\frac{\dd t}t-O_f(L^2N^{-a}).
\end{align*}
Write $\sin^2(\pi t)=\tfrac12-\tfrac12\cos(2\pi t)$. The constant part gives
$\tfrac12L\int_\delta^{1-\delta}F^2$ after $x=\log t/L$. For the cosine part
integrate once by parts: $\phi(t)=F(\log t/L)^2/t$ satisfies
$|\phi|\le C_f/t$ and $|\phi'|\le C_f/t^2$, so it contributes $O_f(N^{-\delta})$.
Thus $\int_0^1|P_u|^2\ge2L^3\int_\delta^{1-\delta}F^2\,(1-o(1))$, and the
same holds on $[-1,0]$ since $|g^\flat|=|g|$. By
Lemma~\ref{lem:endpoint-rep}(c) and the energy estimate,
the difference $p_N-P_u$ has norm $O_f(L)=o(L^{3/2})$. Hence
\[
 \liminf_{N\to\infty}\frac{\norm{p_N}^2}{L^3}\ge4\int_\delta^{1-\delta}F^2 ,
\]
and we let $\delta\downarrow0$.
In particular $p_N\ne0$ for large $N$.

\emph{Conclusion.} For every admissible $f$ the variational principle gives
\[
 \limsup_{N\to\infty}L^2\lambda_-(N,N-1)\le\frac{\pi^2}2\,\frac{\norm f^2}{\norm{Vf}^2}.
\]
Choose admissible $f_j\to\cos(\pi x/2)$ in $L^2(0,1)$; as
$V$ is bounded, the quotients tend to $\pi^4/8$ by Lemma~\ref{lem:volterra}.
Every estimate holds for all large integers $N$, so the conclusion holds
along both parities. \qed

\subsection{\texorpdfstring{Proof of \eqref{eq:sharp-finite}}{Proof of the finite reverse bound}}

Let $L\ge16$, $k_0=\lceil L^2\rceil$ and $\ell=\lfloor N/L\rfloor$; then
$1<k_0\le\ell<N/4$ (e.g.\ $e^L/L^3$ increases for $L\ge3$ and
$e^{16}>4\cdot16^3$). Partition the nodes into the \emph{good layers}
$k_0\le d_k\le\ell$, the \emph{microscopic layers} $d_k<k_0$ (including
$k=0,N$), and the \emph{bulk} $d_k>\ell$, and split $u$ accordingly. Since
interpolation is linear and $\norm{u^{\mathrm{group}}}_w\le\norm u_w$, it
suffices to bound each group.

\emph{Good layers.} For $k_0\le k\le\ell$, $z=\pi k/N\le\pi/L$, so
Lemma~\ref{lem:endpoint-rep}(a) gives
$N^2w_k/k\ge\pi^2\sin z/z-3/k\ge\pi^2(1-\pi^2/(6L^2))-3/L^2\ge\pi^2(1-2/L^2)$,
using $9<\pi^2<10$; the reflected layer is the same. Hence
$\norm g_h^2+\norm{g^\flat}_h^2\le\norm u_w^2/(\pi^2(1-2/L^2))$. By
$\sin\theta\dd\theta\le\pi^2t\dd t/N^2$ and Lemma~\ref{lem:discrete-primitive}
with $T=N/2$,
\[
 \norm{P_u}^2\le4\int_0^{N/2}\sin^2(\pi t)\bigl(|S|^2+|S^\flat|^2\bigr)\frac{\dd t}t
 \le(2\alpha_J^2+6)\bigl(\norm g_h^2+\norm{g^\flat}_h^2\bigr)
 \le\sigma_N^2\norm u_w^2 .
\]
With Lemma~\ref{lem:endpoint-rep}(c),
$\norm{\mathcal I_Nu}\le(\sigma_N+80+6\sqrt{L+1})\norm u_w$.

\emph{Microscopic layers.} For interior indices $1\le d_k<k_0$, the bound
$\norm g_h^2+\norm{g^\flat}_h^2\le\norm u_w^2$ of
Lemma~\ref{lem:endpoint-rep}(c) and Cauchy--Schwarz give
$|S(t)|^2\le H_{k_0}\norm g_h^2$, with $S=0$ for $t<1$. Since
$\int_1^{N/2}\dd t/t\le L$, $\norm{P_u}\le2\sqrt{LH_{k_0}}\norm u_w$, and the
interpolant has norm at most $(2\sqrt{LH_{k_0}}+80+6\sqrt{L+1})\norm u_w$.
For the endpoint $k=0$, $c_0=2$ and the near term of $Y_u$ is
$a_0\sin(\pi t)/(\pi t)$; its squared $L^2(0,1)$ norm is at most
$|a_0|^2N^{-2}\int_0^{N/2}\sin^2(\pi t)\dd t/t\le(L+\pi^2/2)|u_0|^2N^{-2}
\le(L+5)w_0|u_0|^2$ (use $|\sin\pi t|\le\pi t$ on $(0,1)$ and $w_0\ge N^{-2}$).
The far term and the remainder obey the bounds in the proof of
Lemma~\ref{lem:endpoint-rep}(c), since $c_k\ge1$. With the reflected
endpoint, the two endpoint columns contribute at most
$(\sqrt{L+5}+6\sqrt{L+1})\norm u_w\le10\sqrt{L+5}\,\norm u_w$. Altogether the
microscopic layers contribute at most
$(2\sqrt{LH_{k_0}}+80+20\sqrt{L+5})\norm u_w$.

\emph{Bulk.} For $u$ supported in $\ell<k<N-\ell$, discrete orthogonality of
$\cos(l\theta_k)$ under the trapezoidal rule with endpoint half weights gives
$\int_0^\pi|(\mathcal I_Nu)(\cos\theta)|^2\dd\theta=(\pi/N)\sum_k|u_k|^2/c_k-
(\pi/2)|b_N|^2$, where $b_N$ is the top Chebyshev coefficient. As
$\dd x=\sin\theta\dd\theta\le\dd\theta$ and, for bulk indices,
$w_k\ge\sin\theta_k/N\ge2d_k/N^2>2/(LN)$,
\[
 \norm{\mathcal I_Nu}^2\le\frac\pi N\sum_k|u_k|^2\le\frac{\pi L}2\norm u_w^2 .
\]

\emph{Assembly.} Adding the three group bounds, the coefficient of
$\norm u_w$ is at most $\sigma_N$ plus
$160+2\sqrt{LH_{k_0}}+6\sqrt{L+1}+20\sqrt{L+5}+\sqrt{\pi L/2}\le\tau_N$.
Applied to $u_k=p(x_k)$ for $p\in\Pn_N$, this is \eqref{eq:sharp-finite}. \qed

\section{Proofs for the endpoint upper limit}\label{app:edge}

Throughout, $\operatorname{sinc}_1(z)=\sin z/z$ with value one at zero
(the unscaled sinc, distinct from the $\operatorname{sinc}$ of
Appendix~\ref{app:sharp}), $a_j=\pi j$, and the microscopic variable is
$\xi=N\theta$. We use the matrices $M,R,Y,Z$ of
Theorem~\ref{thm:five-thirds} on the indices $0,\dots,N-1$, written
$M_N,R_N,Y_N,Z_N$ when the dependence on $N$ matters.

\subsection{\texorpdfstring{Proof of Proposition~\ref{prop:fixed-trial}}{Proof of the fixed-trial proposition}}

\emph{Exact profiles.} By the cardinal formula of
Theorem~\ref{thm:cardinal},
\[
 \ell_0(\cos\theta)=\frac{\sin(N\theta)\cot(\theta/2)}{2N},\qquad
 \ell_j(\cos\theta)=\frac{(-1)^j\sin\theta\sin(N\theta)}
 {2N\sin\frac{\theta+\theta_j}2\sin\frac{\theta-\theta_j}2}.
\]
Using $\sin\theta=2\sin\frac\theta2\cos\frac\theta2$ and
$\sin\frac{\theta+\theta_j}2\sin\frac{\theta-\theta_j}2
=\sin^2\frac\theta2-\sin^2\frac{\theta_j}2$, one finds
\[
 p_{N,j}(\cos\theta)=\frac{(-1)^j\sin(N\theta)\cos\frac\theta2\,\sin^2\frac{\theta_j}2}
 {N\sin\frac{\theta+\theta_j}2\sin\frac{\theta-\theta_j}2\sin\frac\theta2}.
\]
At $\theta=\xi/N$ this is the exact factorization
\begin{equation}\label{eq:pNj-factor}
 p_{N,j}(\cos(\xi/N))=\chi_j(\xi)\,\varrho_{N,j}(\xi),\qquad
 \varrho_{N,j}(\xi)=\frac{\operatorname{sinc}_1^2\bigl(\frac{a_j}{2N}\bigr)
   \cos\frac\xi{2N}}
  {\operatorname{sinc}_1\bigl(\frac{\xi+a_j}{2N}\bigr)
   \operatorname{sinc}_1\bigl(\frac{\xi-a_j}{2N}\bigr)
   \operatorname{sinc}_1\bigl(\frac\xi{2N}\bigr)} .
\end{equation}
For fixed $\xi$, $\varrho_{N,j}(\xi)\to1$. If $N\ge2j$ and
$0\le\xi\le N\pi$, the three arguments in the denominator have modulus at
most $3\pi/4$, $\pi/2$ and $\pi/2$, so the denominator is at least
$(2\sqrt2/(3\pi))(2/\pi)^2$ and
$0\le\varrho_{N,j}\le3\pi^3/(8\sqrt2)<9$.

\emph{Continuous mass.} The substitution $x=\cos(\xi/N)$ gives
\[
 N^2\int_{-1}^1p^2=\int_0^{N\pi}\xi\operatorname{sinc}_1(\xi/N)
 \Bigl(\sum_ju_j\chi_j(\xi)\varrho_{N,j}(\xi)\Bigr)^2\dd\xi .
\]
From $\chi_j(\xi)=2\int_0^1[\cos(a_js)-(-1)^j]\cos(\xi s)\dd s$ we get
$|\chi_j|\le4$, and from \eqref{eq:chi}, $|\chi_j(\xi)|\le3a_j^2/\xi^3$ for
$\xi\ge2a_j$. Hence the integrand is dominated by the integrable function
$81\,\xi(\sum_j|u_j||\chi_j|)^2$, and dominated convergence gives the
limit $\mathsf M(g)$.

\emph{Discrete mass.} The nodal values of $p_{N,j}$ are $1$ at $x_j$,
$-2(-1)^j$ at $x_0$ and zero elsewhere, so
$N^2Q_N(p^2)=\sum_jN^2w_ju_j^2+4N^2w_0(\sum_j(-1)^ju_j)^2$. The endpoint
weight satisfies $|N^2w_0-1|\le1/(N^2-1)$. For $j\ge1$ write
$b=2\pi j$ and $w_j=(2/N)S_N(\theta_j)$ with the bracket of
Lemma~\ref{lem:identities}. For odd $N=2q+1$, \eqref{eq:telescope} gives
\[
 NS_N(\theta_j)=1+\sum_{r=1}^q\frac{2N}{4r^2-1}\Bigl(1-\cos\frac{br}N\Bigr),
\]
a Riemann sum for $1+\int_0^{1/2}F(x)\dd x$ with
$F(x)=(1-\cos bx)/(2x^2)$: indeed $0\le F\le b^2/4$,
$|F'|\le b^3/12$, and the summands differ from $F(r/N)/N$ by at most
$b^2/[4N(4r^2-1)]$. The even case is the same with an additional top term
$O(1/N)$. Integration by parts gives
$1+\int_0^{1/2}F=(-1)^j+\pi j\operatorname{Si}(\pi j)$, so $N^2w_j\to\omega_j$
for both parities. Hence $N^2Q_N(p^2)\to\mathsf D(g)$.

\emph{Positivity and the bounds.} If $\mathsf M(g)=0$, then
$\sum_ju_j\chi_j\equiv0$, and sampling at $\pi k$ gives $u=0$. Applying
Theorem~\ref{thm:five-thirds} to $p\in\Pn_{N-1}$ and letting
$N\to\infty$ gives $\mathsf D\le\frac53\mathsf M$ on $\mathcal V_0$, so
$\Lambda_{\mathrm{edge}}\le5/3$. Since every $p$ above lies in
$\Pn_{N-1}$, $\lambda_+(N,N-1)\ge N^2Q_N(p^2)/(N^2\int p^2)$, and letting
$N\to\infty$ and then taking the supremum over $g$ gives
$\liminf\lambda_+(N,N-1)\ge\Lambda_{\mathrm{edge}}$.

For $\Lambda_{\mathrm{edge}}\ge1$, take $g=\zeta_j$, whose quotient is
$(\omega_j+4)/\mathsf M(\zeta_j)$. As $j\to\infty$,
$\omega_j/a_j\to\pi$ by $\operatorname{Si}(x)\to\pi/2$. For the mass, put
$\xi=a_j+v$; on $|v|\le a_j/2$,
$\xi\chi_j(\xi)^2=4a_j^4\sin^2v/[\xi(\xi+a_j)^2v^2]$, whose ratio to
$a_j\sin^2v/v^2$ is bounded and tends to one for fixed $v$. Outside this
interval the mass is $o(a_j)$: on $\xi\le a_j/2$ use
$|\chi_j(\xi)|\le C\min(1,1/\xi)$ (mass $O(\log a_j)$), and on
$\xi\ge3a_j/2$ use $|\chi_j|\le Ca_j^2/\xi^3$ (mass $O(1)$). Dominated
convergence gives $\mathsf M(\zeta_j)/a_j\to\int_{\R}(\sin v/v)^2\dd v=\pi$.
Hence the quotients tend to one. \qed

\subsection{The form domain and the continuum identity}

Let $\mathcal V=\{g\in L^2(0,1):\mathsf M(g)<\infty\}$ and define
\[
 \mathsf Y(g)=\int_0^\infty y\Bigl|\int_0^1g(s)e^{-(1-s)y}\dd s\Bigr|^2\dd y,\qquad
 \mathsf Z(g)=\int_0^1\!\!\int_0^1z(s+r)g(s)g(r)\dd s\dd r,
\]
with $z(v)=\max(v,2-v)^{-2}$ on $[0,2]$, so that $0<z\le1$.

\begin{lemma}\label{lem:core-dense}
$\mathcal V_0$ is dense in $\mathcal V$ for the norm
$(\norm g_{L^2}^2+\mathsf M(g))^{1/2}$.
\end{lemma}

\begin{proof}
Extend $g\in\mathcal V$ evenly to $G(s)=g(|s|)$ on $(-1,1)$, zero outside.
Its Fourier transform is $2\widehat g$, so $\mathsf M(g)$ is one eighth of
the squared $L^2(|\xi|\dd\xi)$ norm of $\widehat G$. For $0<r<1$ the dilate
$G(s/r)$ has transform $r\widehat G(r\xi)$; dilation is an isometry of
$L^2(|\xi|\dd\xi)$ and strongly continuous there (check it on continuous
functions supported in compact annuli, then approximate), and also strongly
continuous in $L^2$. Convolving the dilate with an even smooth mollifier
supported inside $(-1,1)$ multiplies $\widehat G$ by a function bounded by
one that tends to one pointwise as the support shrinks, so dominated
convergence gives approximation in both norms by even smooth functions
vanishing near $\pm1$. On $[0,1]$ such a function has a cosine series
$\sum_{j\ge0}b_j\cos(\pi js)$ with rapidly decreasing coefficients; since
it vanishes at $s=1$, it equals $\sum_{j\ge1}b_j[\cos(\pi js)-(-1)^j]$, and
the partial sums converge in $C^2[0,1]$. For a $C^2$ difference $h$ with
$h(1)=0$, two integrations by parts give
$|\widehat h(\xi)|\le\norm h_1$ and
$|\widehat h(\xi)|\le(|h'(0)|+|h'(1)|+\norm{h''}_1)\xi^{-2}$, so
$\mathsf M(h)\to0$.
\end{proof}

\begin{lemma}\label{lem:continuum}
The forms $\mathsf D$ and $\mathsf Y$ are finite and continuous on
$\mathcal V$; $\mathsf M(g)\ge\norm g_{L^2}^2$, $|\mathsf Z(g)|\le\mathsf M(g)$,
$\mathsf Y(g)\le3\mathsf M(g)$, and for every $g\in\mathcal V$,
\begin{equation}\label{eq:continuum}
 \mathsf D(g)=\mathsf M(g)+\tfrac12\bigl[\mathsf Z(g)-\mathsf Y(g)\bigr],
 \qquad
 \tfrac12\bigl[\mathsf Z(g)-\mathsf Y(g)\bigr]\le(\Lambda_{\mathrm{edge}}-1)\mathsf M(g).
\end{equation}
Moreover $\Lambda_{\mathrm{edge}}\le3/2$.
\end{lemma}

\begin{proof}
Let $g=\sum_ju_j\zeta_j\in\mathcal V_0$ and $p=\sum_ju_jp_{N,j}$. The
Chebyshev coefficients of $\ell_k$ are $(2\gamma_k/N)\gamma_i\cos(i\theta_k)$
(proof of Lemma~\ref{lem:identities}), so $Np=\sum_{i<N}a_iT_i$ with
$a_i=g(i/N)$ for $i\ge1$ and $a_0=g(0)/2$. Thus
$a^TM_Na=N^2\int p^2\to\mathsf M(g)$ and $a^TR_Na=N^2Q_N(p^2)\to\mathsf D(g)$
by Proposition~\ref{prop:fixed-trial}. The rescaled entries
$N^2(Z_N)_{ij}=[\max(s,2-s)^2-N^{-2}]^{-1}$, $s=(i+j)/N$, converge uniformly
to $z$ on equal-parity pairs, which form half of all pairs; hence
$a^TZ_Na\to\frac12\mathsf Z(g)$. Similarly
$N^2(Y_N)_{ij}=[(u+v)^2-N^{-2}]^{-1}$ with $u=1-i/N$, $v=1-j/N\ge1/N$,
and $(u+v)^2-N^{-2}\ge\frac34(u+v)^2$. As $g(1)=g'(1)=0$, $|g(s)|\le C(1-s)^2$,
so the weighted kernel $g(1-u)g(1-v)/(u+v)^2$ is bounded and vanishes at the
corner; Riemann sums then give
$a^TY_Na\to\frac12\int\!\!\int g(s)g(r)(2-s-r)^{-2}=\frac12\mathsf Y(g)$, by
$\int_0^\infty ye^{-cy}\dd y=c^{-2}$. The row $i=0$ changes these sums by
$O(1/N)$. The identity $R=M-Y+Z$ gives \eqref{eq:continuum} on
$\mathcal V_0$.

Lemma~\ref{lem:floor} gives $a^TM_Na\ge N^{-1}\sum_ia_i^2\to\norm g^2$, so
$\mathsf M\ge\norm g^2\ge|\mathsf Z|$ on $\mathcal V_0$. Since
$\mathsf D\ge0$, \eqref{eq:continuum} gives $\mathsf Y\le2\mathsf M+\mathsf Z
\le3\mathsf M$; also $\mathsf D\le\Lambda_{\mathrm{edge}}\mathsf M$. Each form
is continuous for the norm of Lemma~\ref{lem:core-dense}; by that lemma they
extend continuously to $\mathcal V$, and the extensions are the displayed
integrals (for $\mathsf Y$: $L^2$ convergence gives pointwise convergence of
the Laplace transforms, and the limit in $L^2(y\dd y)$ agrees; for $\mathsf D$:
$\widehat g$ converges pointwise, and the samples converge in the weighted
sequence space). The relations extend to $\mathcal V$. Finally
$\mathsf Y\ge0$ and $\mathsf Z\le\mathsf M$ give $\mathsf D\le\frac32\mathsf M$.
\end{proof}

\subsection{\texorpdfstring{Proof of Theorem~\ref{thm:edge-limit}}{Proof of the endpoint limit theorem}}

By Proposition~\ref{prop:fixed-trial} it suffices to show that
$\limsup_N\lambda_+(N,N-1)$ is at most $\Lambda_{\mathrm{edge}}$. Let $\Lambda^*$ be a
subsequential limit; it is finite by Theorem~\ref{thm:five-thirds}. If
$\Lambda^*\le1$ there is nothing to prove, since $\Lambda_{\mathrm{edge}}\ge1$.
Otherwise let $a=a^{(N)}$ be the coefficient vector of a maximizing
polynomial $p_N=\sum_ia_iT_i$ with $a^TM_Na=\int p_N^2=1$, so that
$a^TR_Na=\lambda_+(N,N-1)$.

\emph{Endpoint profiles.} Lemma~\ref{lem:floor} gives
$N^{-1}\sum_ia_i^2\le1$. Let $e_N$ and $o_N$ be the step functions equal to
$a_{2k}$ and $a_{2k+1}$ on $[2k/N,\min\{(2k+2)/N,1\})$ (with $a_N=0$), so
$\norm{e_N}^2+\norm{o_N}^2\le2$, and put $g_N^\pm=(e_N\pm o_N)/2$. Along a
further subsequence $g_N^\pm\rightharpoonup g^\pm$ weakly in $L^2(0,1)$.
For every $v\in C^1[0,1]$,
\begin{equation}\label{eq:profile-weak}
 \frac1N\sum_ia_iv(i/N)\to\int_0^1g^+v,\qquad
 \frac1N\sum_i(-1)^ia_iv(i/N)\to\int_0^1g^-v :
\end{equation}
the cell quadrature errors are bounded by $\norm{v'}_\infty$ times
$N^{-1}(N^{-1}\sum a_i^2)^{1/2}$, and for odd $N$ the final half cell costs
$O(\norm v_\infty N^{-1/2})$ because $|a_{N-1}|\le\sqrt N$.

\emph{Mass.} With $v(s)=\cos(\xi s)$, \eqref{eq:profile-weak} gives
$N^{-1}p_N(\cos(\xi/N))\to\widehat{g^+}(\xi)$ and
$N^{-1}p_N(-\cos(\xi/N))\to\widehat{g^-}(\xi)$ for each $\xi$, and these
quantities are bounded by $(N^{-1}\sum_ia_i^2)^{1/2}\le1$. For fixed $T$,
the two disjoint endpoint pieces of $\int p_N^2$ are
\[
 \int_0^T\xi\operatorname{sinc}_1(\xi/N)\,\bigl|N^{-1}p_N(\pm\cos(\xi/N))\bigr|^2\dd\xi .
\]
Dominated convergence, then $T\to\infty$, gives
\begin{equation}\label{eq:profile-mass}
 \mathsf M(g^+)+\mathsf M(g^-)\le1 ;
\end{equation}
in particular $g^\pm\in\mathcal V$.

\emph{The compact part.} The rescaled kernel of $Z_N$ converges uniformly to
$z(s+r)$, which is uniformly continuous, so the cell replacement changes the
operator norm by $o(1)$ (with $O(N^{-1/2})$ from a final half cell). The
integral operator with kernel $z(s+r)$ is compact. Since equal-parity pairs
give $\frac14[\mathsf Z(e_N)+\mathsf Z(o_N)]=\frac12[\mathsf Z(g_N^+)+\mathsf Z(g_N^-)]$
up to $o(1)$, weak convergence yields
\[
 a^TZ_Na\to\tfrac12\bigl[\mathsf Z(g^+)+\mathsf Z(g^-)\bigr].
\]

\emph{The positive part.} With
$E^{\mathrm e}_N(y)=\frac2N\sum_{i\ \mathrm{even}}a_ie^{-(1-i/N)y}$ and
$E^{\mathrm o}_N$ the odd analogue, the substitution $x=e^{-y/N}$ in the
integral representation of $Y$ (proof of Theorem~\ref{thm:five-thirds}) gives
the exact formula
\[
 a^TY_Na=\int_0^\infty\frac N4\sinh\frac yN\,
 \bigl[E^{\mathrm e}_N(y)^2+E^{\mathrm o}_N(y)^2\bigr]\dd y .
\]
By \eqref{eq:profile-weak}, $E^{\mathrm e}_N(y)\to\int(g^++g^-)e^{-(1-s)y}\dd s$
and $E^{\mathrm o}_N(y)\to\int(g^+-g^-)e^{-(1-s)y}\dd s$ for each $y$, and
$\frac N4\sinh\frac yN\ge\frac y4$. Fatou's lemma gives
\[
 \liminf a^TY_Na\ge\tfrac12\bigl[\mathsf Y(g^+)+\mathsf Y(g^-)\bigr].
\]
No convergence of the unbounded matrix $Y_N$ itself is needed.

\emph{Conclusion.} By $R_N=M_N-Y_N+Z_N$, \eqref{eq:continuum} and
\eqref{eq:profile-mass},
\begin{align*}
 \Lambda^*-1=\lim a^T(Z_N-Y_N)a
 &\le\tfrac12\sum_\pm\bigl[\mathsf Z(g^\pm)-\mathsf Y(g^\pm)\bigr]\\
 &\le(\Lambda_{\mathrm{edge}}-1)\sum_\pm\mathsf M(g^\pm)
 \le\Lambda_{\mathrm{edge}}-1 .
\end{align*}
The bound $\Lambda_{\mathrm{edge}}\le3/2$ is Lemma~\ref{lem:continuum}. \qed

\section{The certificate for Proposition~\ref{prop:edge-certified}}\label{app:certificate}

\subsection{Closed forms of the model mass}

Write $a_j=\pi j$, $s_j=(-1)^j$ and
$\operatorname{Cin}(x)=\int_0^x(1-\cos v)\,v^{-1}\dd v$.

\begin{lemma}\label{lem:closed-form}% IDs: PD-T548
For $i,j\ge1$ the integrals $C_{ij}=\int_0^\infty\xi\chi_i(\xi)\chi_j(\xi)\dd\xi$
converge absolutely, and
\[
 C_{ij}=\frac{2s_is_j\bigl(a_i^2\operatorname{Cin}(2a_j)-a_j^2\operatorname{Cin}(2a_i)\bigr)}{a_i^2-a_j^2}\quad(i\ne j),\qquad
 C_{jj}=2\bigl[a_j\operatorname{Si}(2a_j)+\operatorname{Cin}(2a_j)\bigr].
\]
Hence $\mathsf M(\sum_ju_j\zeta_j)=\sum_{i,j}u_iC_{ij}u_j$ for finite $u$.
\end{lemma}

\begin{proof}
By \eqref{eq:chi}, $\chi_j$ is continuous with removable singularities,
bounded on compact sets and $O_j(\xi^{-3})$, so $\xi\chi_i\chi_j$ is
absolutely integrable. Let $a$ be a positive integer multiple of $\pi$ and
$\mathcal H(x)=\int_0^x\sin^2v\,v^{-1}\dd v=\operatorname{Cin}(2x)/2$. Put
\[
 I(a)=\int_0^\infty\frac{\sin^2t}{t(t^2-a^2)}\dd t,\qquad
 K(a)=\int_0^\infty\frac{t\sin^2t}{(t^2-a^2)^2}\dd t ,
\]
both absolutely convergent. For $R>a$, partial fractions give
$a^2\int_0^R\frac{\sin^2t}{t(t^2-a^2)}\dd t=\frac12\int_0^R\frac{\sin^2t}{t-a}\dd t
+\frac12\int_0^R\frac{\sin^2t}{t+a}\dd t-\mathcal H(R)$. Since $\sin^2$ has
period $\pi$ and $\sin^2v/v$ is odd, translation gives
$\int_0^R\sin^2t\,(t\mp a)^{-1}\dd t=\mathcal H(R\mp a)-\mathcal H(a)$, so
the right side equals $\frac12\mathcal H(R-a)+\frac12\mathcal H(R+a)-\mathcal H(R)-\mathcal H(a)$,
and the first three terms tend to zero as $R\to\infty$, being bounded by
$\frac12\log\frac R{R-a}+\frac12\log\frac{R+a}R$. Thus
$I(a)=-\operatorname{Cin}(2a)/(2a^2)$. For $i\ne j$, the identity
$[(t^2-a^2)(t^2-b^2)]^{-1}=[(t^2-a^2)^{-1}-(t^2-b^2)^{-1}]/(a^2-b^2)$ gives
$C_{ij}=4s_is_ja^2b^2[I(a)-I(b)]/(a^2-b^2)$ with $a=a_i$, $b=a_j$, which is
the first formula. For the diagonal, integration by parts
($\frac{\dd}{\dd t}(t^2-a^2)^{-1}=-2t(t^2-a^2)^{-2}$, with vanishing boundary
terms at $0$, at $a^\pm$ and at infinity) gives
$K(a)=\frac12\int_0^\infty\sin(2t)(t^2-a^2)^{-1}\dd t$, and the same
translation argument with $\sin(2v)/v$ even gives this integral as
$\operatorname{Si}(2a)/a$. Finally
$t^{-1}(t^2-a^2)^{-2}=a^{-4}[t^{-1}-t(t^2-a^2)^{-1}]+a^{-2}t(t^2-a^2)^{-2}$,
with the bracket kept together at each cutoff, gives
$C_{jj}=4a^2[K(a)-I(a)]=2[a\operatorname{Si}(2a)+\operatorname{Cin}(2a)]$.
\end{proof}

\subsection{The finite reduction}

By \eqref{eq:Lambda-edge}, for every nonzero $u\in\R^J$,
\[
 \Lambda_{\mathrm{edge}}\ge q(u):=\frac{u^TDu}{u^TCu},\qquad
 D=\operatorname{diag}(\omega_1,\dots,\omega_J)+4ss^T,
\]
with $C=(C_{ij})_{i,j\le J}$ from Lemma~\ref{lem:closed-form} and
$s=(s_j)$. We take $J=512$ and a vector $u$ of binary rationals, stored
exactly as hexadecimal binary64 strings. It was selected by a floating-point
generalized eigensolver; the selection plays no role in the proof, because
the bound holds for every $u$.

The check encloses $u^TDu$ and $u^TCu$ in intervals with endpoints on the
grid $2^{-256}\mathbb Z$, using unbounded integers and rounding every product
and quotient outward. The scalar inputs are enclosed as follows.
\begin{itemize}
\item $\pi$: Machin's formula $\pi=16\arctan\frac15-4\arctan\frac1{239}$ with
alternating arctangent series.
\item $\operatorname{Si}(x)$ for $x=n\pi$, $n<16$, and $\operatorname{Cin}(x)$
for $x\le16\pi$: the entire series
\[
 \operatorname{Si}(x)=\sum_{k\ge0}\frac{(-1)^kx^{2k+1}}{(2k+1)(2k+1)!},\qquad
 \operatorname{Cin}(x)=\sum_{k\ge1}\frac{(-1)^{k+1}x^{2k}}{(2k)(2k)!},
\]
summed until
the ratios of successive absolute terms are decreasing and at most $1/2$; the
tail is then bounded by twice the next term.
\item $x=n\pi$ with $n\ge16$: with
$F(x)=\int_0^\infty e^{-xt}(1+t^2)^{-1}\dd t$ and
$G(x)=\int_0^\infty te^{-xt}(1+t^2)^{-1}\dd t$ one has
$\operatorname{Si}(x)=\frac\pi2-\cos x\,F(x)-\sin x\,G(x)$ and
$\operatorname{Ci}(x)=\sin x\,F(x)-\cos x\,G(x)$ (differentiate, using
$F'=-G$, $G'=F-1/x$, and compare limits at $0$ and $\infty$), so
$\operatorname{Si}(n\pi)=\frac\pi2-(-1)^nF(n\pi)$ and
$\operatorname{Ci}(n\pi)=-(-1)^nG(n\pi)$. Expanding $(1+t^2)^{-1}$ to
$20$ terms leaves remainders of sign $(-1)^{20}$ bounded by $40!/x^{41}$ and
$41!/x^{42}$.
\item $\operatorname{Cin}(2\pi j)$ for $j\ge8$: from
$\operatorname{Cin}(x)=\gamma+\log x-\operatorname{Ci}(x)$, the identity
$\operatorname{Cin}(2\pi j)=\operatorname{Cin}(16\pi)+\log(j/8)+\operatorname{Ci}(16\pi)-\operatorname{Ci}(2\pi j)$,
in which Euler's constant cancels; logarithms of rationals by power-of-two
reduction and $\log q=2\sum_kz^{2k+1}/(2k+1)$, $z=(q-1)/(q+1)\le1/3$, with a
geometric tail bound.
\end{itemize}

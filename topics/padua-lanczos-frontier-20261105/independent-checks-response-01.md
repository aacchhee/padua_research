from pathlib import Path
from fractions import Fraction as F
import json
T = Path('topics/padua-lanczos-frontier-20261105/evidence')
def plus(a,b): return a[0]+b[0], a[1]+b[1]
def minus(a,b): return a[0]-b[1], a[1]-b[0]
def times(a,c): return tuple(sorted(x*c for x in a))
def total(v,c):
    z=(F(0),F(0))
    for a,b in zip(v,c): z=plus(z,times(a,b))
    return z
def weights(ns):
    m=F(sum(ns),len(ns)); d=sum((F(n)-m)**2 for n in ns)
    return {n:(F(n)-m)/d for n in ns}
for grid in ['triangle','rotated']:
    o=json.loads((T/f'g1-compact-{grid}.json').read_text()); S=o['scale']
    assert [(r[0],r[1]) for r in o['rows']]==[
        (n,r) for n in range(2,21) for r in ['bj','mirror','alt','pivot']]
    d={(n,r):((F(ll,S),F(lh,S)),(F(ul,S),F(uh,S)))
       for n,r,ll,lh,ul,uh in o['rows']}
    A=[]; B=[]
    for n in range(2,21):
        q=[d[n,r] for r in ['mirror','alt','pivot']]
        low=tuple(min(x[0][j] for x in q) for j in [0,1])
        high=tuple(min(x[1][j] for x in q) for j in [0,1])
        A.append(minus(d[n,'bj'][0],high))
        B.append(minus(d[n,'bj'][1],low))

    e=[we.get(n,F(0)) for n in range(2,21)]

    y=[b if z>0 else a if z<0 else times(plus(a,b),F(1,2))
       for a,b,z in zip(A,B,c)]
    early=total([a if z>=0 else b for a,b,z in zip(A,B,e)],e)
    mid=total([times(plus(a,b),F(1,2)) for a,b in zip(A,B)],c)
    margin=total(y,c)
    assert margin[0]>0 and early[0]>0 and mid[0]>0
    print(grid,margin,early,mid)
